## 1. 가입상품 생성 쿼리 #
-- 장기 TM 에서 판매중인 상품 리스트 가져오기
DROP TABLE GDREC_TMGD_LIST PURGE;
CREATE TABLE GDREC_TMGD_LIST  AS (	
SELECT 
	DISTINCT 
	A.*
	, CASE WHEN B.GDNM LIKE '%Young%' THEN 'L04' -- 자녀
           WHEN B.GDNM LIKE '%더건강 더실속%' OR B.MKTG_GD_CSFCD in ('L02','L01')					THEN 'L01' -- 종합
            WHEN GDNM LIKE '%더건강%' THEN 'L01'
           WHEN B.GDNM LIKE '%시그니처 여성 건강%'	THEN 'L01'        -- 종합
      	   WHEN B.MKTG_GD_CSFCD = 'L03'							THEN 'L03' -- 운전자
      	   WHEN B.MKTG_GD_CSFCD = 'L04'							THEN 'L04' -- 자녀
      	   WHEN B.MKTG_GD_CSFCD = 'L06'							THEN 'L06' -- 암
      	   WHEN B.MKTG_GD_CSFCD = 'L11'							THEN 'L11' -- 재물
      	   WHEN B.MKTG_GD_CSFCD = 'L08'							THEN 'L08' -- 간병
      	   WHEN B.GDNM LIKE '%세이프투게더%'						THEN 'SAFE' -- 세이프투게더
           WHEN B.GDNM LIKE '%실손의료보험%'						THEN 'SS' -- 실손
      	   WHEN REGEXP_LIKE(B.GDNM, '참 편한|WELL100|간편|3N5|355')  THEN 'SI'
      	   WHEN B.MKTG_GD_CSFCD in ('L12','L13')				THEN 'L123' --연저축가입자수가 부족해서 기타로 분류
      	   ELSE 'ETC' END AS GD_TYPE
    , B.GDNM
    , D.DTCNM AS SL_CHNCD
FROM
(SELECT * FROM IGD_GD_SL_TRM
WHERE SL_NDDT >= SYSDATE
AND GDCD LIKE '%LA%') A
LEFT JOIN M_CRM_GD_CSF_INFO B
ON 0=0
	AND A.GDCD = B.GDCD
LEFT JOIN IGD_GD_SL_CHN_REL C
ON 0=0
	AND A.GDCD = C.GDCD
LEFT JOIN SL_CHNCD D
ON 0=0
	AND C.SL_CHNCD = D.SL_CHNCD
WHERE 1=1
    AND D.DTCNM = 'TM'
    AND A.GDCD IN (SELECT DISTINCT GDCD FROM TB_LNG_BJONG_MASTER WHERE SALE_END_YN = 'Y')
);


-- 상품/담보별 가입 연령 및 필수담보여부
DROP TABLE GDREC_TMGD_LIST_INFO PURGE;
CREATE TABLE GDREC_TMGD_LIST_INFO  AS (	
SELECT 
	DISTINCT 
	A.*
	, B.CVRCD 
	, B.CVR_PRSNM  -- 담보명
	, B.CVR_CLA_CN  -- 담보약관내용
	, B.CVR_BA_TRT_FLGCD -- 특약여부
	, B.ISAMT_NEED_YN -- 가입금액필요여부
	, B.MN_IS_AGE  -- 최소가입연령
	, B.MX_IS_AGE  -- 최대가입연령
	, B.IS_AV_SEXCD  -- 성별코드
	, C.ESSN_IS_YN  -- 필수가입여부
	, C.MN_IS_AV_AGE
	, C.MX_IS_AV_AGE
	, C.ISAMT_MNVL  -- 최소가입금액
	, C.ISAMT_MXVL  -- 최대가입금액
	, C.REC_CVR_YN  -- 추천담보여부
	, CASE WHEN F.GDCD IS NOT NULL THEN 1 ELSE 0 END AS 무사고_YN
FROM SDADAM.GDREC_TMGD_LIST A
LEFT JOIN IGD_GD_CVR B
ON 0=0
	AND A.GDCD = B.GDCD
LEFT JOIN INS_GD_CE_PL_CVR C
ON 0=0
	AND A.GDCD = C.GDCD
	AND B.CVRCD = C.CVRCD 
LEFT JOIN IGD_GD_CVR_ACM_RK D
ON 0=0
	AND B.CVRCD = D.CVRCD
LEFT JOIN IGD_GD_US_ATR F
ON 0=0
  AND F.GDCD = A.GDCD
  AND F.ATRCD         ='PA00119'
	);

-- 장기 TM 보유고객 가져오기
DROP TABLE M_BIZ_CTM_INFO_CVR_LIST_RECT_T02 PURGE;
CREATE TABLE M_BIZ_CTM_INFO_CVR_LIST_RECT_T02  AS
(
SELECT  A2.CLS_YYMM
	   ,A2.PLYNO                     AS PLYNO                                   /*증권번호*/
       ,A2.CTMNO                     AS CTMNO                                   /*고객번호(장기피보험자)*/
       ,A2.RELPC_SEQNO               AS RELPC_SEQNO                             /*관계자순번*/
       ,A2.STFNO                     AS STFNO                                   /*직원번호*/
       ,A2.USRNO                     AS USRNO                                   /*사용인번호*/
       ,A2.GDCD                      AS GDCD                                    /*상품코드*/
       ,A2.INS_ST                    AS INS_ST                                  /*보험시기*/
       ,A2.INS_CLSTR                 AS INS_CLSTR                               /*보험종기*/
       ,A2.BRTH                      AS BRTH
       ,CASE WHEN F_ISDATE(A2.BRTH) = 'T'
                  THEN (CASE WHEN A2.CTM_TPCD = '01'                           /*고객구분코드:개인*/
                     THEN  EXTRACT(YEAR FROM to_yminterval(AGE(ADD_MONTHS(F_LAST_DAY(A2.CLS_YYMM), +6), TO_DATE(A2.BRTH, 'YYYYMMDD'))  DEFAULT '000-00' ON CONVERSION ERROR ))
                       - (CASE WHEN EXTRACT(MONTH FROM to_yminterval(AGE(ADD_MONTHS(F_LAST_DAY(A2.CLS_YYMM), +6), TO_DATE(A2.BRTH,'YYYYMMDD') ) DEFAULT '000-00' ON CONVERSION ERROR) ) = 1
                           AND SUBSTR(ADD_MONTHS(F_LAST_DAY(A2.CLS_YYMM), +6) ,6,2) = SUBSTR(A2.BRTH,6,2)
                                                  THEN 1                       /*해당월의 해당일자까지 비교후 출생일이전이면 -1을 셋팅*/
                                                  ELSE 0                       /*해당월이 아닐경우 기본값으로만 처리하기위해  0을 셋팅*/
                                        END)
                        END)
        END                AS INS_AGE    
        
        
FROM (SELECT  /*+ FULL(A1) FULL(B1) LEADING(A1) */
			  T.CLS_YYMM
             , A1.PLYNO                          AS PLYNO                        /*증권번호*/
             ,B1.CTMNO                          AS CTMNO                        /*고객번호(장기피보험자)*/
             ,B1.RELPC_SEQNO                    AS RELPC_SEQNO                  /*관계자순번*/
             ,A1.DH_STFNO                       AS STFNO                        /*직원번호*/
             ,COALESCE(A1.DH_USR_NO,'9999999')  AS USRNO                        /*사용인번호*/
             ,A1.GDCD                           AS GDCD                         /*상품코드*/
             ,A1.INS_ST                         AS INS_ST                       /*보험시기*/
             ,A1.INS_CLSTR                      AS INS_CLSTR                    /*보험종기*/
             ,ROW_NUMBER() OVER(PARTITION BY A1.PLYNO                           /*증권번호*/
                                            ,B1.CTMNO                           /*고객번호*/
                                            ,T.CLS_YYMM
                                    ORDER BY B1.RELPC_SEQNO) AS RN              /*관계자순번*/
            ,F_BIRTH_DAY_DEC(B1.CTM_DSCNO) AS BRTH 
            ,C.CTM_TPCD
      FROM 
      (SELECT 
        	A.*
        	, B.GDNM_CLEAN	
        FROM GDREC_ORIGIN_INS_CR_PRED A
        LEFT JOIN M_CRM_REC_GDNM_TM B
        ON 0=0
        	AND A.GDCD = B.GDCD
        WHERE 1=1 AND TO_CHAR(A.INS_ST,'YYYYMM') = F_PREV_MONTH(A.CLS_YYMM,'1')
        ) A1                                        /**월별보유계약**/
	  INNER JOIN GDRES_SYS_YYMM_PRED T 
	  ON A1.CLS_YYMM = T.CLS_YYMM
      INNER JOIN
          INS_CR_RELPC          B1                                        /**계약관계자**/
      ON   1 = 1
      AND  A1.PLYNO                  = B1.PLYNO                                 /*증권번호*/
      AND  B1.IKD_GRPCD              IN ('LA')                                  /*보종군코드:장기보험(종피 포함)*/
      AND  B1.RELPC_STCD             IN ('01','04')                             /*관계자상태코드:정상,납입면제*/
      AND  B1.RELPC_TPCD             = '02'                                     /*관계자유형코드:피보험자*/
      AND  B1.VALD_NDS_YN            = '1'                                      /*유효배서여부*/
	    AND  B1.NDS_AP_STR_DTHMS <= T.CLS_NDDT
	    AND  B1.NDS_AP_ND_DTHMS > T.CLS_NDDT      
      AND  B1.CTMNO                  IS NOT NULL                                /*고객번호 존재건*/
    LEFT JOIN CUS_CTM C 
      ON 0=0
        AND B1.CTMNO = C.CTMNO
      WHERE 1=1
      AND  A1.IKD_GRPCD              = 'LA'                                     /*보종군코드:장기보험*/
-----      AND b1.RELPC_SEQNO NOT IN (607,467,262,259,256,253,174,138,50,37,11,10,7,5,3,2)
     ) A2
WHERE  1=1
AND  A2.RN = 1
) ;
-- 담보 가입금액 추출
DROP TABLE M_BIZ_CTM_INFO_CVR_LIST_RECT_T03 PURGE;
CREATE TABLE M_BIZ_CTM_INFO_CVR_LIST_RECT_T03  AS
(
SELECT /*+ FULL(A1) FULL(B1) FULL(C1) LEADING(A1) */
        A1.CLS_YYMM
       ,A1.PLYNO                     AS PLYNO                                   /*증권번호*/
       ,B1.CVRCD                     AS CVRCD                                   /*담보코드*/
       ,B1.CVR_SEQNO                 AS CVR_SEQNO                               /*담보순번*/
       ,A1.CTMNO                     AS CTMNO                                   /*고객번호*/
       ,A1.RELPC_SEQNO               AS RELPC_SEQNO                             /*관계자순번*/
       ,A1.STFNO                     AS STFNO                                   /*직원번호*/
       ,A1.USRNO                     AS USRNO                                   /*사용인번호*/
       ,A1.INS_ST                    AS INS_ST                                  /*보험시기*/
       ,A1.INS_CLSTR                 AS INS_CLSTR                               /*보험종기*/
       ,A1.GDCD                      AS GDCD                                    /*상품코드*/
       ,A1.BRTH                      AS BRTH
       ,A1.INS_AGE                   AS INS_AGE
       ,B1.CVR_STCD                  AS CVR_STCD                                /*담보상태코드*/
       ,B1.INS_ST                    AS CVR_INS_ST                              /*담보보험시기*/
       ,B1.INS_CLSTR                 AS CVR_INS_CLSTR                           /*담보보험종기*/
       ,F_DIFF_DATE('MM',TO_CHAR(B1.INS_CLSTR ,'YYYYMMDD'),TO_CHAR(F_LAST_DAY(A1.CLS_YYMM),'YYYYMMDD'))
                           AS CVR_CLSTR_ST_RNDMC                                /*담보종기기준경과월수(현재일자~담보보험종기의 월수)*/
       ,F_DIFF_DATE('MM',TO_CHAR(B1.INS_ST ,'YYYYMMDD'),TO_CHAR(F_LAST_DAY(A1.CLS_YYMM),'YYYYMMDD'))
                           AS CVR_ST_ST_RNDMC     
       ,B1.AP_PRM                    AS AP_PRM                                  /*적용보험료*/
       ,B1.ISAMT                     AS ISAMT                                   /*가입금액*/
       ,B1.AUTO_RNW_AV_YN            AS AUTO_RNW_AV_YN                          /*자동갱신가능여부*/
       ,B1.AUTO_RNW_CVR_CNLDT        AS AUTO_RNW_CVR_CNLDT                      /*자동갱신담보해지일자*/
       ,B1.RNW_NDDT                  AS RNW_NDDT                                /*갱신종료일자*/
       ,C1.MDCS_RGT_BJ_YN            AS MDCS_RGT_BJ_YN                          /*의료비등록대상여부*/
       ,C1.RLPMI_CVR_YN              AS RLPMI_CVR_YN                            /*실손담보여부*/

FROM  M_BIZ_CTM_INFO_CVR_LIST_RECT_T02  A1                                  /**월별보유계약정보**/
INNER JOIN (SELECT T.CLS_YYMM
                  , B1.*
                  ,ROW_NUMBER() OVER (PARTITION BY T.CLS_YYMM, PLYNO, RELPC_OJ_SEQNO, CVRCD, CVR_SEQNO ORDER BY NDSNO DESC) RN
                FROM  
                (
                SELECT 
                	B1.CVR_STCD
                	, B1.INS_ST
                	, B1.INS_CLSTR
                	, B1.AP_PRM
                	, B1.ISAMT
                	, B1.AUTO_RNW_AV_YN
                	, B1.AUTO_RNW_CVR_CNLDT
                	, B1.RNW_NDDT
                	, B1.RELPC_OJ_SEQNO
                	, B1.PLYNO
                	, B1.CVRCD
                	, B1.CVR_SEQNO
                	, B1.NDS_AP_STR_DTHMS
                	, B1.NDS_AP_ND_DTHMS
                	, B1.NDSNO
                FROM 
    			INS_CR_CVR B1
    			WHERE 1=1
    			    AND EXISTS (SELECT 1 FROM M_BIZ_CTM_INFO_CVR_LIST_RECT_T01 A2 WHERE A2.THCP_CVRCD = B1.CVRCD)
    			    AND  B1.IKD_GRPCD                = 'LA'   
    				AND  B1.VALD_NDS_YN              = '1'                                          /*유효배서여부*/
    				AND  B1.CVR_STCD                IN ('01','07','08')                             /*담보상태코드:정상,완납,납입면제*/
    				AND  B1.CVR_BJ_FLGCD            IN ('01','03')                                  /*담보대상구분코드:인담보,적립담보*/
    				) B1     
                INNER JOIN GDRES_SYS_YYMM_PRED T 
    	            ON 0=0                                                /**계약담보**/
                	  AND  B1.NDS_AP_STR_DTHMS <= T.CLS_NDDT
    	              AND  B1.NDS_AP_ND_DTHMS > T.CLS_NDDT
            ) B1
ON   1 = 1
AND  A1.PLYNO                    = B1.PLYNO                                     /*증권번호*/
AND  A1.RELPC_SEQNO              = B1.RELPC_OJ_SEQNO                            /*관계자목적물순번*/
AND  B1.INS_CLSTR               >= F_LAST_DAY(B1.CLS_YYMM)                                   /*담보보험종기 >= 기준일자*/
AND  B1.RN=1

LEFT OUTER JOIN
     IGD_GD_CVR    C1                                                     /**상품담보**/
ON   1 = 1
AND  B1.CVRCD                    = C1.CVRCD                                     /*담보코드*/
AND  A1.GDCD                     = C1.GDCD                                      /*상품코드*/
AND  C1.AP_NDDT                  >  F_LAST_DAY(B1.CLS_YYMM)                                   /*적용종료일자*/
AND  C1.AP_STRDT                 <= F_LAST_DAY(B1.CLS_YYMM)

) ;

--가입금액 보정
DROP TABLE M_BIZ_CTM_INFO_CVR_LIST_RECT_T06 PURGE;
CREATE TABLE M_BIZ_CTM_INFO_CVR_LIST_RECT_T06  AS
(
SELECT  A2.*
       ,(CASE WHEN A2.THCP_IBAMT_STNM IN ('가입금액')                           /*당사보험금액기준명=가입금액*/
               AND A2.THCP_GNAMT_ITCD IN ('JH0004','JH0005','JH0006','JH0007','JH0008','JH0009')
                                                             THEN A2.ISAMT                                                                /*후유장해 6개담보에 대해서만 실제가입금액*/
              WHEN A2.THCP_IBAMT_STNM IN ('가입금액')        THEN (A2.ISAMT / A2.THCP_INS_MULT_STAMT) * A2.THCP_GNAMT                     /*지급금액 = (실제가입금액/당사보험배수기준금액) * 당사보장금액*/
              WHEN A2.THCP_IBAMT_STNM IN ('정액')            THEN A2.THCP_GNAMT * 10000                                                   /*지급금액 = 고정된 금액지급(단위:만원, 따라서 10000 곱)*/
              WHEN A2.THCP_IBAMT_STNM IN ('보험료','일시납') THEN ((A2.AP_PRM / 10000) / A2.THCP_INS_MULT_STAMT) * A2.THCP_GNAMT * 10000  /*지급금액 = ((회당납입보험료/10000) / 당사보험배수기준금액) * 당사보장금액*10000*/
                                                             ELSE 0
         END) * A2.PYCT    AS IS_CVAMT                                          /*가입금액*/

 FROM (SELECT /*+ FULL(A1) */
               A1.*
              ,CASE WHEN A1.THCP_AN_ND_DTNM IN ('보증')                         /*당사연금만료세부명=보증*/
                     AND A1.THCP_AN_TRM_NM  IN ('만기') THEN                    /*당사연금기간명=만기*/
                         (CASE WHEN A1.CVR_CLSTR_ST_RNDMC < A1.THCP_AN_PY_CT    /*담보종기기준경과월수 < 당사연금지급횟수*/
                                    THEN A1.THCP_AN_PY_CT                       /*당사연금지급횟수*/
                                    ELSE A1.CVR_CLSTR_ST_RNDMC                  /*담보종기기준경과월수*/
                          END)
                    WHEN A1.THCP_AN_ND_DTNM IN ('확정')  THEN A1.THCP_AN_PY_CT  /*당사연금만료세부명=확정*/
                                                         ELSE 1
               END            AS PYCT                                           /*보장금액을 계산용*/
              ,CASE WHEN A1.INS_ST < F_LAST_DAY(A1.CLS_YYMM)  THEN                         /*보험시기 < 현재일자*/
                         (CASE WHEN A1.CVR_ST_ST_RNDMC + 1 BETWEEN A1.THCP_INSTR_MNVL AND A1.THCP_INSTR_MXVL
                                    THEN 'Y'
                                    ELSE 'N'
                          END)
                    ELSE
                         (CASE WHEN A1.CVR_ST_ST_RNDMC BETWEEN A1.THCP_INSTR_MNVL AND A1.THCP_INSTR_MXVL
                                    THEN 'Y'
                                    ELSE 'N'
                          END)
               END            AS FLGCD_1                                        /*경과기간 체크*/
              ,CASE WHEN A1.INS_AGE BETWEEN A1.THCP_INS_AGE_MNVL AND A1.THCP_INS_AGE_MXVL
                                  THEN 'Y'
                                  ELSE 'N'
               END            AS FLGCD_2                                        /*보험연령 체크*/

       FROM M_BIZ_CTM_INFO_CVR_LIST_RECT_T04   A1
       WHERE 1=1
       AND  A1.THCP_INSTR_CSFNM   = '만기'                                      /*당사보험기간분류명=만기*/
       AND  A1.INS_ST             IS NOT NULL                                   /*보험시기 존재건*/
      ) A2
WHERE 1 = 1
AND  A2.FLGCD_1     = 'Y'                                                       /*담보시기기준경과월수가 당사보험기간최소,최대값내 존재건*/
AND  A2.FLGCD_2     = 'Y'                                                       /*보험연령이 당사보험연령최소,최대값내 존재건*/
AND  A2.PYCT        > 0                                                         /*보장금액을 계산용 지급횟수 > 0*/
) ;

-- 현재시점 기준 날자세팅
DROP TABLE GDRES_SYS_YYMM_PRED PURGE;
CREATE TABLE GDRES_SYS_YYMM_PRED AS (
            SELECT
                YYMM AS CLS_YYMM
                ,F_ADD_MONTHS_A(F_LAST_DAY(YYMM),-2) AS CLS_YYMM_2MB
                ,F_ADD_MONTHS_A(F_LAST_DAY(YYMM),-1) AS CLS_YYMM_1MB
                ,FIRST_DAY(YYMM) AS CLS_STRDT
                ,LEAST(SYSDATE, YYMM_NDDT) AS CLS_NDDT
                ,LEAST(SYSDATE, TO_DATE(TO_CHAR(YYMM_NDDT, 'YYYY-MM-DD') || ' 23:59:59', 'YYYY-MM-DD HH24:MI:SS')) AS CLS_NDDT_TS
                ,LEAST(CAST(F_LAST_DAY(YYMM) AS DATE), SYSDATE) AS CLS_NDDT_LD
            FROM M_COD_YYMM
            WHERE YYMM = TO_CHAR(SYSDATE,'YYYYMM')
            );

-- 사고일자 지급금 (장기 , 자동차)
DROP TABLE GDREC_ORIGIN_INS_CLM_PRED PURGE;
CREATE TABLE GDREC_ORIGIN_INS_CLM_PRED AS
SELECT
	T.CLS_YYMM
	,T.CLS_NDDT
	,A.MN_NRDPS_CTMNO
	,CASE WHEN B.IKD_GRPCD IN ('FA','MA') THEN 'FAMA' ELSE B.IKD_GRPCD END AS IKD_GRPCD
	,B.CLMDT
	,SUM(B.WONCR_CV_PYAMT) AS PY_AMT
FROM GDRES_SYS_YYMM_PRED T
INNER JOIN GDREC_ORIGIN_INS_CR_PRED A
ON 0=0
	AND T.CLS_YYMM = A.CLS_YYMM
LEFT JOIN M_CMP_GN_CVRCL_PY_IBNF_LIST B
ON 0=0
	AND A.MN_NRDPS_CTMNO = B.CLMPS_CTMNO
WHERE 0=0
	AND B.CLMDT BETWEEN F_LAST_DAY(F_PREV_MONTH(T.CLS_YYMM,'60')) AND T.CLS_NDDT
GROUP BY
	T.CLS_YYMM
	,T.CLS_NDDT
	,A.MN_NRDPS_CTMNO
	,CASE WHEN B.IKD_GRPCD IN ('FA','MA') THEN 'FAMA' ELSE B.IKD_GRPCD END 
	,B.CLMDT

UNION

SELECT
	CLS_YYMM
	,CLS_NDDT
	,MN_NRDPS_CTMNO
	,IKD_GRPCD
	,CLMDT
	,SUM(RPBL_PY_IBAMT) + SUM(OPTN_PY_IBAMT) AS PY_AMT
FROM
	(
	SELECT DISTINCT
		T.CLS_YYMM
		,T.CLS_NDDT
		,A.MN_NRDPS_CTMNO
		,'CA' AS IKD_GRPCD 
		,C.CLMDT
		,B.RPBL_PY_IBAMT
		,B.OPTN_PY_IBAMT
	FROM GDRES_SYS_YYMM_PRED T
	INNER JOIN GDREC_ORIGIN_INS_CR_PRED A
	ON 0=0
		AND T.CLS_YYMM = A.CLS_YYMM
	LEFT JOIN M_CMP_CR_CLM_CRST B
	ON 0=0
		AND A.MN_NRDPS_CTMNO = B.NRDPS_CTMNO
	LEFT JOIN M_CMP_CR_CMP_CLM_CR C
	ON 0=0
		AND B.RCPDT = C.RCPDT
		AND B.RCP_NV_SEQNO = C.RCP_NV_SEQNO
	WHERE 0=0
		AND C.CLMDT BETWEEN F_LAST_DAY(F_PREV_MONTH(T.CLS_YYMM,'60')) AND T.CLS_NDDT
	) AA
GROUP BY
	CLS_YYMM
	,CLS_NDDT
	,MN_NRDPS_CTMNO
	,IKD_GRPCD
	,CLMDT
;

-- 상품별 설계 횟수
DROP TABLE GDREC_ORIGIN_PL_ALL_PRED PURGE;
CREATE TABLE GDREC_ORIGIN_PL_ALL_PRED  AS (	
SELECT 
  *
FROM
  (SELECT 
    CLS_YYMM
    ,CRT_CTMNO
    , MN_NRDPS_CTMNO
    ,GD_TYPE
    ,COUNT(DISTINCT PLNO) AS PLNO_CNT
  FROM
    (
		SELECT 
			A.*
			, B.PLDT
			, B.PL_STCD
			, B.GD_TYPE
			, B.PLNO
			, RANK() OVER (PARTITION BY A.CRT_CTMNO, A.MN_NRDPS_CTMNO ORDER BY CLS_YYMM DESC) AS RN
		FROM GDREC_ORIGIN_INS_PRMCNT_PRED A
		LEFT JOIN GDREC_ORIGIN_PL_PRED B
		ON 0=0
			AND A.CRT_CTMNO = B.CRT_CTMNO
			AND F_LAST_DAY(A.CLS_YYMM) >= B.PLDT
			AND A.MN_NRDPS_CTMNO = B.MN_NRDPS_CTMNO
      )
  WHERE RN = 1
  GROUP BY 
    CLS_YYMM
    ,CRT_CTMNO
    , MN_NRDPS_CTMNO
    ,GD_TYPE)
  PIVOT
  (
  	MAX(PLNO_CNT)
  	FOR GD_TYPE IN ('L01' AS L01,'L03' AS L03,'L04' AS L04,'L06' AS L06,'L08' AS L08,'SAFE' AS SAFE , 'SS' AS SS, '유병자'  AS 유병자,'L11' AS L11,'L123' AS L123)
  ) P
  );

CREATE TABLE GDREC_ORIGIN_CLM_G_PRED AS (


  SELECT 
    A.CLS_YYMM
    , A.CLMPS_CTMNO
    , A.clmps_ctm_dscno
    , SUM(A.WONCR_CV_PYAMT) 지급보험금
    , MAX(A.PRVDT) 최근사고일자
  FROM (
        SELECT
          BB.CLS_YYMM 
          ,AA.* 
        FROM M_CMP_GN_CVRCL_PY_IBNF_LIST AA
        INNER JOIN GDREC_ORIGIN_INS_PRMCNT_PRED BB
        ON 0=0
          AND AA.CLMPS_CTMNO = BB.CRT_CTMNO 
          AND F_LAST_DAY(BB.CLS_YYMM) >= AA.PRVDT
        ) 	A 		 /* TBL : 일반담보별지급보험금명세 */
  LEFT OUTER JOIN LSE_DGN 			C		/* TBL : 진단 (RCP_YYMM,RCP_NV_SEQNO,CLMPS_SEQNO,CLM_NV_SEQNO,DGNCD) */
  ON 1=1
  AND A.RCP_YYMM = C.RCP_YYMM					/*접수년월*/
  AND A.RCP_NV_SEQNO = C.RCP_NV_SEQNO			/*접수조사순번*/
  AND A.CLMPS_SEQNO = C.CLMPS_SEQNO			/*사고자순번*/
  AND A.CLM_NV_SEQNO = C.CLM_NV_SEQNO			/*사고조사순번*/
  
  LEFT OUTER JOIN SUS_DSACD E
  ON REPLACE(C.DGNCD,'*','') = E.SUS_DSACD
  
  WHERE 1=1
  	AND A.PLYNO IS NOT NULL
  	AND A.IKD_GRPCD = 'LA'
  	AND A.CCL_YN='0'
  	AND COALESCE(A.UDRTK_TYCD,'01') != '03' 				/* 타사간사 제외 */
  	AND A.XI_PY_FLGCD IN ('02','03')					/* 추산지급,환입 */
  	AND A.CLM_NV_CSFCD = '11'							/* 인보장 */
  	AND A.CLM_CVR_LCLCD IN ('G001') -- 사망/후유장해/입원/수술
  GROUP BY A.CLS_YYMM, A.CLMPS_CTMNO,A.clmps_ctm_dscno
  )
;

DROP TABLE GDREC_TMGD_CLM_INFO_PRED;
CREATE TABLE GDREC_TMGD_CLM_INFO_PRED  AS (	
SELECT 
	A.CLS_YYMM
	, A.CRT_CTMNO
	, A.MN_NRDPS_CTMNO
	, TRUNC(MONTHS_BETWEEN(F_LAST_DAY(A.CLS_YYMM) , B1.최근사고일자)/12,0) AS 입원청구
	, TRUNC(MONTHS_BETWEEN(F_LAST_DAY(A.CLS_YYMM) , B2.DGNDT)/12,0) AS 입원고지
	, TRUNC(MONTHS_BETWEEN(F_LAST_DAY(A.CLS_YYMM) , B3.DGNDT)/12,0) AS 중대질환고지
	, TRUNC(MONTHS_BETWEEN(F_LAST_DAY(A.CLS_YYMM) , B4.DT)/12,0) AS 중대질환청구
	, TRUNC(MONTHS_BETWEEN(F_LAST_DAY(A.CLS_YYMM) , B5.최근사고일자)/12,0) AS 무사고
FROM GDREC_ORIGIN_INS_PRMCNT_PRED A
LEFT JOIN GDREC_ORIGIN_CLM_G_PRED B1
ON 0=0
	AND A.CLS_YYMM = B1.CLS_YYMM
	AND A.MN_NRDPS_CTMNO = B1.CLMPS_CTMNO
LEFT JOIN GDREC_ORIGIN_CLM_HSP_PRED B2
ON 0=0
	AND A.CLS_YYMM = B2.CLS_YYMM
	AND A.MN_NRDPS_CTMNO = B2.MN_NRDPS_CTMNO
LEFT JOIN GDREC_ORIGIN_CLM_MTT_PRED B3
ON 0=0
	AND A.CLS_YYMM = B3.CLS_YYMM
	AND A.MN_NRDPS_CTMNO = B3.MN_NRDPS_CTMNO
LEFT JOIN GDREC_ORIGIN_CLM_CSF_PRED B4
ON 0=0
	AND A.CLS_YYMM = B4.CLS_YYMM
	AND A.MN_NRDPS_CTMNO = B4.CTMNO
LEFT JOIN GDREC_ORIGIN_CLM_ALL_PRED B5
ON 0=0
	AND A.CLS_YYMM = B5.CLS_YYMM
	AND A.MN_NRDPS_CTMNO = B5.CLMPS_CTMNO
	);

	, B1.L03_설계완료
	, B1.L04_설계완료
	, B1.L06_설계완료
	, B1.L08_설계완료

	, B1.L123_설계완료

		ELSE TO_CHAR(B2.입원고지) END AS 입원고지

		WHEN B2.중대질환고지 IS NULL THEN 'X'
		ELSE TO_CHAR(B2.중대질환고지) END AS 중대질환고지

		WHEN B2.중대질환청구 IS NULL THEN 'X'
		ELSE TO_CHAR(B2.중대질환청구) END AS 중대질환청구
	, CASE WHEN B2.무사고 > 10 THEN '10'
		WHEN B2.무사고 IS NULL THEN 'X'
		ELSE TO_CHAR(B2.무사고) END AS 무사고

	AND A.MN_NRDPS_CTMNO = B1.MN_NRDPS_CTMNO

ON 0=0
	AND A.CRT_CTMNO = B2.CRT_CTMNO
	AND A.CLS_YYMM = B2.CLS_YYMM
	AND A.MN_NRDPS_CTMNO = B2.MN_NRDPS_CTMNO);
