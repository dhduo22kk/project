/* 기준연월 SETTING */
DROP TABLE TBL_DS_YYMM;
CREATE TABLE TBL_DS_YYMM AS (
            SELECT
                YYMM AS CLS_YYMM
                ,F_ADD_MONTHS_A(F_LAST_DAY(YYMM),-6) AS CLS_YYMM_6MB
                ,F_ADD_MONTHS_A(F_LAST_DAY(YYMM),-2) AS CLS_YYMM_2MB
                ,F_ADD_MONTHS_A(F_LAST_DAY(YYMM),-1) AS CLS_YYMM_1MB
                ,FIRST_DAY(YYMM) AS CLS_STRDT
                ,LEAST(SYSDATE, YYMM_NDDT) AS CLS_NDDT
                ,LEAST(SYSDATE, TO_DATE(TO_CHAR(YYMM_NDDT, 'YYYY-MM-DD') || ' 23:59:59', 'YYYY-MM-DD HH24:MI:SS')) AS CLS_NDDT_TS
                ,LEAST(CAST(F_LAST_DAY(YYMM) AS DATE), SYSDATE) AS CLS_NDDT_LD
            FROM M_COD_YYMM
            WHERE YYMM = TO_CHAR(SYSDATE,'YYYYMM')
            );

/*1-1. 장기 TM 대상자 보유 고객 추출 쿼리 - 장기 TM의 계약을 보유한 고객이면 대상 */
DROP TABLE TBL_DS_CRLIST PURGE;
CREATE TABLE TBL_DS_CRLIST NOLOGGING AS(
SELECT
  	LCA.CLS_YYMM
  	, 'LA'			AS IKD_GRPCD
  	, LCA.PLYNO
  	, LCA.CNRDT
  	, LCA.CRT_CTMNO	AS CTMNO
  	, LCA.DH_STFNO
  FROM HWMRT.M_LCR_MTHY_LTINS_CR LCA
  INNER JOIN SDADAM.TBL_DS_YYMM T ON T.CLS_YYMM = LCA.CLS_YYMM
  INNER JOIN M_BIZ_MTHY_PS_CR PCA
  ON 0=0
    AND 	LCA.CLS_YYMM	= PCA.CLS_YYMM
    AND 	LCA.PLYNO		= PCA.PLYNO
  WHERE 1=1
  	AND LCA.DH_STFNO IN (SELECT STFNO FROM M_ORG_BZ_ORGN WHERE HDQT_ORGCD = '1300012' AND BRNM LIKE '%장기%' AND STF_FLGCD = '02'));


DROP TABLE TBL_DS_CUSLIST PURGE;
CREATE TABLE TBL_DS_CUSLIST NOLOGGING AS(
SELECT
DISTINCT 
    LCB.CLS_YYMM
    , LCB.IKD_GRPCD
    , LCB.PLYNO
    , LCB.CNRDT
    , LCB.CTMNO
    , LCB.DH_STFNO
    , LPA.CMS_CTMNO
    , MCA.PRS_TM_CTMNO  
	  , MCA.FNL_MKTG_AGRYN										-- 최종마케팅동의여부
	  , MCA.MKTG_TL_RCV_YN										-- 마케팅전화수신여부
	  , MCA.FNL_MKTG_AGRE_NDDT                -- 마케팅동의종료일자
FROM TBL_DS_CRLIST LCB
LEFT JOIN HWMRT.M_CMS_LGSYS_CTM_CMS_MPP LPA
ON 0=0
  AND  LCB.CTMNO = LPA.CTMNO
LEFT JOIN HWMRT.M_CMS_CTM_CLS MCA
ON 0=0
	AND LCB.CLS_YYMM = MCA.CLS_YYMM
	AND LPA.CMS_CTMNO = MCA.CMS_CTMNO
WHERE 1=1
  AND LPA.CMS_CTMNO IS NOT NULL
  AND PCA.PLYNO IS NOT NULL)
;

/*1-2. 장기 TM 대상자 가망 고객 추출 쿼리 - TM 마동 고객이거나 1년내 캠페인 배정 이력 있으면 가망고객  */
DROP TABLE TBL_DS_PSSLIST PURGE;
CREATE TABLE TBL_DS_PSSLIST NOLOGGING AS (
SELECT DISTINCT *
FROM
    (
         SELECT                                   -- 2. 마케팅동의(전화)가 유효하고, TB_CUSTOMER 등록 가망고객
           B.CLS_YYMM
           , B.INP_DTHMS
           , A.CMS_CTMNO
           , A.PRS_CTMNO AS CTMNO
           , A.PRS_TM_CTMNO TM_CUSTID
           , B.INP_USR_ID  
           , '비대면등록_CUSTID' AS INP_SEG
      	   , A.FNL_MKTG_AGRYN										-- 최종마케팅동의여부
      	   , A.MKTG_TL_RCV_YN										-- 마케팅전화수신여부
      	   , A.FNL_MKTG_AGRE_NDDT                -- 마케팅동의종료일자
        FROM HWMRT.M_CMS_CTMCS A
        INNER JOIN 
        	(SELECT	
        		T.CLS_YYMM
        		, A.*
        	FROM SDADAM.TBL_DS_YYMM T
        	INNER JOIN TB_CUSTOMER A
        	ON 0=0
        		AND A.INP_DTHMS <= T.CLS_NDDT_TS
        	WHERE INP_USR_ID IN (SELECT STFNO FROM M_ORG_BZ_ORGN WHERE HDQT_ORGCD = '1300012')
        		) B
        ON 1=1
        	AND A.PRS_TM_CTMNO = B.CUSTID        
    	    AND A.FNL_MKTG_AGRYN = 1
    	    AND A.MKTG_TL_RCV_YN = 1
    
    UNION ALL                               

        SELECT                                   -- 3. 마케팅동의(전화)가 유효하고, 직전1년이내 캠페인으로 배정
        	DISTINCT                           
        	  T.CLS_YYMM
        	  , X2.ASDT AS INP_DTHMS
        	  , X2.CMS_CTMNO
            , X2.PRS_CTMNO AS CTMNO
            , X2.PRS_TM_CTMNO TM_CUSTID
            , CMPG_ALLCT_STFNO AS INP_USR_ID
        	  , '캠페인배정' AS INP_SEG
      	    , X3.FNL_MKTG_AGRYN										-- 최종마케팅동의여부
      	    , X3.MKTG_TL_RCV_YN										-- 마케팅전화수신여부
      	    , X3.FNL_MKTG_AGRE_NDDT                -- 마케팅동의종료일자
        FROM SDADAM.TBL_DS_YYMM T
        INNER JOIN HWMRT.M_CMS_CTM_CLS X1 /*CMS고객마감*/
        ON T.CLS_YYMM = X1.CLS_YYMM
        INNER JOIN HWMRT.M_CMS_INR_CHN_CMPG_DB_CLS X2 /*CMS캠페인DB현황*/
        ON X1.CMS_CTMNO = X2.CMS_CTMNO /*CMS고객번호*/
        AND X1.CLS_YYMM = X2.CLS_YYMM
        AND X2.ASDT >= CLS_NDDT_TS - INTERVAL '1' YEAR 
        INNER JOIN HWMRT.M_CMS_CTMCS X3
        ON X1.CMS_CTMNO = X3.CMS_CTMNO
        AND X3.FNL_MKTG_AGRYN = 1
        AND X3.MKTG_TL_RCV_YN = 1
        WHERE CMPG_LCLCD IN ('C')
        )
    WHERE CMS_CTMNO NOT IN (SELECT DISTINCT CMS_CTMNO FROM SDADAM.TBL_DS_CUSLIST)
);

CREATE TABLE TBL_DS_CMPG AS 
SELECT
    	DISTINCT
    	 B.TASKID 
    	, B3.TASKNAME 
    	, B.LISTGROUPID 
    	, B4.LISTNAME     리스트그룹
    	, B.LISTID  
    	, B5.LISTNAME     리스트
    	, B.SPONSORCODE
    	, B.REGISTDATE 
    	, B.ALLOCDATE  
    	, B.ALLOCCHARGERID
    	, B.CUSTID
    	, B2.CTMNO
    	, C.CMS_CMPG_ID
    	, C.CMS_CMPG_NM
    FROM TB_LIST B
    LEFT JOIN  TB_CUSTOMER B1 ON B.CUSTID = B1.CUSTID
    LEFT JOIN  CUS_CTM B2 ON B1.CUST_DSCNO = B2.CTM_DSCNO
    LEFT JOIN  TB_TASK B3  ON B.TASKID = B3.TASKID
    LEFT JOIN  TB_LIST_GROUP B4 ON B.LISTGROUPID = B4.LISTGROUPID
    LEFT JOIN  TB_LIST_CAT B5 ON B.LISTID = B5.LISTID
    LEFT JOIN  TBL_DS_CMPG_LIST C
    ON C.LIST_NO = B.LISTID
      AND C.TASK_NO = B.TASKID
      AND C.LIST_GPNO = B.LISTGROUPID
    WHERE 1=1
      AND NVL(B.DELYN,'N')        <> 'Y'
      AND B.ALLOCCHARGERID IS NOT NULL
      AND B.CUSTID IN (SELECT DISTINCT PRS_TM_CTMNO FROM TBL_DS_CTM_LIST);

/* 전화안내방법동의여부 (상품소개동의)                                     */
/* 추출 출처 : 태동일C, 자동차POM_DB                                       */
/***************************************************************************/
--EXEC P_DROP_TABLE ('E_TM_LTRM_POM_EXTR_KDH_T12');
DROP TABLE E_TM_LTRM_POM_EXTR_KDH_T12;
CREATE TABLE E_TM_LTRM_POM_EXTR_KDH_T12
AS
SELECT  A2.CTM_DSCNO                                     /* 고객식별번호   */
      , A2.TL_NTC_MTD_AGRYN                              /* 전화안내방법동의여부 */
      , TRUNC(A2.AP_STR_DTHMS)  AS TL_NTC_MTD_AGR_STR_DT /* 전화안내방법도의시작일자 */
      , TRUNC(A2.AP_ND_DTHMS)   AS TL_NTC_MTD_AGR_ND_DT  /* 전화안내방법도의종료일자 */
FROM (
       SELECT  B2.CTM_DSCNO          AS CTM_DSCNO              /* 고객식별번호   */
             , CASE WHEN A1.INP_DTHMS    >=  TRUNC(F_ADD_MONTHS_GP(CURRENT_DATE, -24))                        /* 금감원기준 2년까지만 상품소개동의가능 */
                    AND  A1.AP_ND_DTHMS  >   TRUNC(CURRENT_DATE)                                              /* 적용종료일시 */
                    AND  A1.AP_STR_DTHMS <=  CAST(TO_CHAR(CURRENT_DATE,'YYYYMMDD')||'235959' AS TIMESTAMP(0)) /* 적용시작일시 */
               THEN '1'
               ELSE '0'
               END                   AS TL_NTC_MTD_AGRYN     /* 전화안내방법동의여부 */
             , A1.AP_STR_DTHMS       AS AP_STR_DTHMS         /* 적용시작일시         */
             , A1.AP_ND_DTHMS        AS AP_ND_DTHMS          /* 적용종료일시         */
             , ROW_NUMBER() OVER (PARTITION BY B2.CTM_DSCNO
                                  ORDER BY     CASE WHEN  A1.INP_DTHMS           >=  TRUNC(F_ADD_MONTHS_GP(CURRENT_DATE, -24))                                     /* 금감원기준 2년까지만 상품소개동의가능 */
                                                    AND   A1.AP_ND_DTHMS         >   TRUNC(CURRENT_DATE)                                                           /* 적용종료일시 */
                                                    AND   A1.AP_STR_DTHMS        <=  CAST(TO_CHAR(CURRENT_DATE,'YYYYMMDD')||'235959' AS TIMESTAMP(0))     /* 적용시작일시 */
                                               THEN '1'
                                               ELSE '0'
                                               END             DESC  /* 전화안내방법동의여부 */
                                             , A1.AP_ND_DTHMS DESC
                                             , A1.AP_STR_DTHMS DESC)   AS RN
       FROM (
             SELECT  DISTINCT
                     B1.CTM_DSCNO
                   , C1.CUSTID
             FROM  E_TM_LTRM_POM_EXTR_KDH_T01 B1
             INNER JOIN TB_CUSTOMER C1              /** 고객마스터 **/
             ON    C1.CUST_DSCNO = B1.CTM_DSCNO
            ) B2
       INNER JOIN TB_CTM_CRDIU_AGRE A1              /** 고객신용정보활용동의 **/
       ON    A1.CUSTID              = B2.CUSTID
       WHERE 1=1
       AND   A1.CRDIU_AGRE_FLGCD     = '03'               /* 신용정보활용동의구분코드  03:상품소개 */
       AND   A1.CRDIU_AGRE_DTL_FLGCD IN( '01','02','03')  /* 신용정보활용동의세부구분코드 01:처리이용,02:조회이용,03:제공이용 */
     ) A2
WHERE RN = 1
;

/*캠페인 팀별 구분 */
SELECT * 
FROM
(SELECT 
	A.CLS_YYMM 
	, CASE WHEN TMNM LIKE '%POM%' THEN '[C02]POM'
		WHEN TMNM LIKE '%웹인%' THEN '[C01]웹인'
		WHEN TMNM LIKE '%미디어%' THEN '[C14]미디어'
		WHEN TMNM LIKE '%O/B%' THEN '[C07]O/B'
		WHEN TMNM LIKE '%혼합%' THEN '[C16]혼합'
		WHEN TMNM LIKE '%직인%' THEN '[C03]직인'
		WHEN TMNM LIKE '%호전환%' THEN '[C09]호전환'
		WHEN TMNM LIKE '%채팅%' THEN '[C08]채팅'
		WHEN TMNM LIKE '%인콜%' THEN '[C18]인콜'
		ELSE NULL END CMPG_MDCCD
	, SUM(CASE WHEN CMPG_LCLCD = 'B' THEN CR_CRCC_RNPRM

						'E01','E02','E04','E05')

		WHEN TMNM LIKE '%웹인%' THEN '[C01]웹인'
		WHEN TMNM LIKE '%미디어%' THEN '[C14]미디어'

    (SELECT 
        A0.기준년월  
      	, A0.DH_STFNO
      	, A0.PLYNO
      	, A1.CTMNO
      	, A2.INS_ST

      	, A0.DH_STFNO
      	, A0.PLYNO
    FROM  INS_INCM_PRM A0
    WHERE 1=1
        AND A0.DP_CASCD           =   '01'           /*입금원인코드:신계약*/
        AND NVL(A0.RTN_FLGCD,'XX') <> '01'
      	AND A0.PPDT BETWEEN CASE WHEN TRUNC(SYSDATE) = TRUNC(SYSDATE ,'MM')
      	                          THEN TRUNC(ADD_MONTHS(SYSDATE,-1),'MM')
      	                          ELSE TRUNC(SYSDATE,'MM') END
      	                   AND SYSDATE

        AND A0.IKD_GRPCD = 'LA') A0
    LEFT JOIN INS_CR_RELPC A1
    ON 1=1
        AND A1.PLYNO = A0.PLYNO
        AND A1.AP_NDDT            =   TO_DATE('99991231', 'YYYYMMDD')
        AND A1.RELPC_TPCD         =   '01'
        AND A1.PRS_RELPC_YN       =   '1'
    LEFT JOIN INS_INS_CR A2
    ON 1=1
        AND A2.PLYNO              =   A0.PLYNO 
        AND A2.AP_NDDT            =   TO_DATE('99991231', 'YYYYMMDD')
        AND A2.PLY_LVL_FLGCD      =   '01'      /*증권레벨구분코드*/
        AND A2.IKD_GRPCD          =   'LA') AA

    (SELECT
    	DISTINCT
    	 B.TASKID 
    	, B3.TASKNAME 
    	, B.LISTGROUPID 
    	, B4.LISTNAME     리스트그룹
    	, B.LISTID  
    	, B5.LISTNAME     리스트
    	, B.SPONSORCODE
    	, B.REGISTDATE 
    	, B.ALLOCDATE  
    	, B.ALLOCCHARGERID
    	, B.CUSTID
    	, B2.CTMNO
    FROM TB_LIST B

  AND AA.DH_STFNO =  BB.ALLOCCHARGERID

DROP TABLE DATABIZ_FUTVAL_CUS_ADD_PRED;
CREATE TABLE DATABIZ_FUTVAL_CUS_ADD_PRED AS
SELECT
  A1.CLS_YYMM
	, A1.CTMNO
	, CTM_SEX
	, CASE WHEN CTM_AGE BETWEEN 18 AND 29 THEN '20대 이하'
		   WHEN CTM_AGE BETWEEN 30 AND 39 THEN '30대'
		   WHEN CTM_AGE BETWEEN 40 AND 49 THEN '40대'
		   WHEN CTM_AGE BETWEEN 50 AND 59 THEN '50대'
		   WHEN CTM_AGE BETWEEN 60 AND 99 THEN '60대 이상'
		   ELSE NULL END AS AGE_CD
	, CTM_AGE
	, SD
	, SGNG
	, PSTNO
	, PSTNO_ID
	,E.JBCD
	,E.JB_CH_SEQNO
	,F.JBNM
FROM (
	SELECT DISTINCT 
		T.CLS_YYMM
		, T.CTMNO
		, T.CTM_SEX
		, T.CTM_AGE
		, A.LACO_DONG_FLGCD
		, A.BLDNM
		, A.SD
		, A.SGNG
		, A.PSTNO
		, A.REF_IT 
		, TO_CHAR(A.AP_STR_DTHMS,'YYYYMM')    AS AP_STR_YYMM
		, TO_CHAR(A.AP_ND_DTHMS,'YYYYMM')     AS AP_ND_YYMM
		, A.AP_STR_DTHMS
		, A.AP_ND_DTHMS
		, COALESCE(B.PSTNO_ID,C.PSTNO_ID, D.PSTNO_ID) AS PSTNO_ID
		, ROW_NUMBER() OVER(PARTITION BY T.CLS_YYMM, T.CTMNO ORDER BY A.AP_STR_DTHMS DESC, CASE WHEN B.PSTNO_ID IS NULL THEN 1 ELSE 0 END, CASE WHEN C.PSTNO_ID IS NULL THEN 1 ELSE 0 END) AS RK
	FROM DATABIZ_FUTVAL_ALL_CUS_INFO_PRED T
	INNER JOIN CUS_CTM_ADR_CRR A 
	ON 0=0
		AND T.CTMNO = A.CTMNO
		AND A.AP_ND_DTHMS > = F_LAST_DAY(T.CLS_YYMM) -- 시점
		AND A.AP_STR_DTHMS < F_LAST_DAY(T.CLS_YYMM) -- 시점
		AND A.ADR_FLGCD = '100' 
	
	LEFT OUTER JOIN CUS_PSTNO_RODNM_BA_ADR B 
	ON 1=1 
		AND A.BLD_ADMNO = B.BLD_ADMNO		
	
	LEFT OUTER JOIN CUS_PSTNO_RODNM_BA_ADR C
	ON 1=1 
		AND A.PSTNO = C.PSTNO
		AND A.SD=C.SD 
		AND A.SGNG = C.SGNG
		AND A.ROAD_NM = C.ROAD_NM
		AND A.MN_BLDNO = C.MN_BLDNO 
		AND COALESCE(A.SUB_BLDNO,'0') = COALESCE(C.SUB_BLDNO,'0') 
	
	LEFT OUTER JOIN CUS_PSTNO_RODNM_BA_ADR D
	ON 1=1 
		AND A.PSTNO = D.PSTNO
		AND A.SD=D.SD 
		AND A.SGNG = D.SGNG
		AND A.ROAD_NM = D.ROAD_NM
		AND A.MN_BLDNO = D.MN_BLDNO
	
	) A1
LEFT  JOIN CUS_CTM_JB_CRR E				
	ON 0=0			
		AND A1.CTMNO = E.CTMNO		
		AND E.AP_STR_DTHMS <= F_LAST_DAY(CLS_YYMM)
		AND E.AP_ND_DTHMS  >  F_LAST_DAY(CLS_YYMM)
		
	LEFT JOIN CUS_JB F		
	ON 0=0			
		AND E.JBCD = F.JBCD		
		AND E.JB_CH_SEQNO = F.JB_CH_SEQNO
WHERE 1=1
	AND RK = 1
;

		       WHEN H.주부 != 1 AND H.학생 != 1 AND H.무직 != 1 THEN  COALESCE(D.LBR_ICR_AVG_ICAMT, D1.LBR_ICR_AVG_ICAMT, F.LBR_ICR_AVG_ICAMT, F1.LBR_ICR_AVG_ICAMT, D2.LBR_ICR_AVG_ICAMT, D3.LBR_ICR_AVG_ICAMT, F2.LBR_ICR_AVG_ICAMT, F3.LBR_ICR_AVG_ICAMT)

		, COALESCE(D.AVG_ICAMT, D1.AVG_ICAMT, F.AVG_ICAMT, F1.AVG_ICAMT, D2.AVG_ICAMT, D3.AVG_ICAMT, F2.AVG_ICAMT, F3.AVG_ICAMT) AS INCOME /*평균소득*/
		, CASE WHEN  H.주부 = 1 THEN COALESCE(D.HMKR_CRDVL_AVG_POCT, D1.HMKR_CRDVL_AVG_POCT, F.HMKR_CRDVL_AVG_POCT, F1.HMKR_CRDVL_AVG_POCT, D2.HMKR_CRDVL_AVG_POCT, D3.HMKR_CRDVL_AVG_POCT, F2.HMKR_CRDVL_AVG_POCT, F3.HMKR_CRDVL_AVG_POCT)
			   WHEN H.학생 = 1 THEN COALESCE(D.STT_CRDVL_AVG_POCT, D1.STT_CRDVL_AVG_POCT, F.STT_CRDVL_AVG_POCT, F1.STT_CRDVL_AVG_POCT, D2.STT_CRDVL_AVG_POCT, D3.STT_CRDVL_AVG_POCT, F2.STT_CRDVL_AVG_POCT, F3.STT_CRDVL_AVG_POCT)
			   WHEN H.무직 = 1 THEN COALESCE(D.OTJOB_CRDVL_AVG_POCT, D1.OTJOB_CRDVL_AVG_POCT, F.OTJOB_CRDVL_AVG_POCT, F1.OTJOB_CRDVL_AVG_POCT, D2.OTJOB_CRDVL_AVG_POCT, D3.OTJOB_CRDVL_AVG_POCT, F2.OTJOB_CRDVL_AVG_POCT, F3.OTJOB_CRDVL_AVG_POCT)
			   WHEN H.전문직 = 1 THEN COALESCE(D.SPJOB_CRDVL_AVG_POCT, D1.SPJOB_CRDVL_AVG_POCT, F.SPJOB_CRDVL_AVG_POCT, F1.SPJOB_CRDVL_AVG_POCT, D2.SPJOB_CRDVL_AVG_POCT, D3.SPJOB_CRDVL_AVG_POCT, F2.SPJOB_CRDVL_AVG_POCT, F3.SPJOB_CRDVL_AVG_POCT)
		       WHEN H.법인대표 = 1 THEN  COALESCE(D.CRP_PRS_CRDVL_AVG_POCT, D1.CRP_PRS_CRDVL_AVG_POCT, F.CRP_PRS_CRDVL_AVG_POCT, F1.CRP_PRS_CRDVL_AVG_POCT, D2.CRP_PRS_CRDVL_AVG_POCT, D3.CRP_PRS_CRDVL_AVG_POCT, F2.CRP_PRS_CRDVL_AVG_POCT, F3.CRP_PRS_CRDVL_AVG_POCT)

		       ELSE NULL END AS CREDIT_SEG /*평균신용점수(NICE) - 직업별*/  	
		, COALESCE(D.AVG_CRD_VLPOC, D1.AVG_CRD_VLPOC, F.AVG_CRD_VLPOC, F1.AVG_CRD_VLPOC, D2.AVG_CRD_VLPOC, D3.AVG_CRD_VLPOC, F2.AVG_CRD_VLPOC, F3.AVG_CRD_VLPOC) AS CREDIT /*평균신용점수(NICE)*/
	FROM SDADAM.DATABIZ_FUTVAL_CUS_ADD_PRED B
    LEFT OUTER JOIN CUS_PSTNO_BAADR_SCT_ID C        /* 우편번호 ID별 250M격자 매칭값 */ 
    ON B.PSTNO_ID = C.PSTNO_ID
    LEFT OUTER JOIN M_CRM_IC_CRD_GRID_ST_STS D   /* 250M격자 기준 개인 단위 통계 */ 
    ON 0=0
        AND C.BAADR_SCT_ID = D.TOT_UNT_GRID_NM
        AND B.CTM_SEX = D.SEXCD
        AND B.AGE_CD = D.AGGB_NM 
        AND D.ST_YYMM = '202312'
    LEFT OUTER JOIN M_CRM_IC_CRD_GRID_ST_STS D1   /*  250M격자 기준 개인 단위 통계 - 성별ALL */
    ON 0=0
        AND C.BAADR_SCT_ID = D1.TOT_UNT_GRID_NM
        AND D1.SEXCD = 'ALL'
        AND B.AGE_CD = D1.AGGB_NM 
        AND D1.ST_YYMM = '202312'

        AND B.CTM_SEX = D2.SEXCD

        AND D3.SEXCD = 'ALL'
        AND D3.AGGB_NM = 'ALL'
        AND D3.ST_YYMM = '202312'

    ON 0=0
        AND B.PSTNO = F.TOT_UNT_PSTNO
        AND B.CTM_SEX = F.SEXCD
        AND B.AGE_CD = F.AGGB_NM
        AND F.ST_YYMM = '202312'
    LEFT OUTER JOIN M_CRM_IC_CRD_PSTNO_ST_STS F1 /*  우편번호 기준 개인 단위 통계 - 성별ALL */ 
    ON 0=0
        AND B.PSTNO = F1.TOT_UNT_PSTNO
        AND F1.SEXCD = 'ALL'
        AND B.AGE_CD = F1.AGGB_NM 
        AND F1.ST_YYMM = '202312'
    LEFT OUTER JOIN M_CRM_IC_CRD_PSTNO_ST_STS F2 /*  우편번호 기준 개인 단위 통계 - 연령ALL */ 
    ON 0=0
        AND B.PSTNO = F2.TOT_UNT_PSTNO
        AND B.CTM_SEX = F2.SEXCD
        AND F2.AGGB_NM = 'ALL'
        AND F2.ST_YYMM = '202312'
    LEFT OUTER JOIN M_CRM_IC_CRD_PSTNO_ST_STS F3 /*  우편번호 기준 개인 단위 통계 - 성별ALL, 연령ALL */ 
    ON 0=0
        AND B.PSTNO = F3.TOT_UNT_PSTNO
        AND F3.SEXCD = 'ALL'
        AND F3.AGGB_NM = 'ALL'
        AND F3.ST_YYMM = '202312' 
	LEFT OUTER JOIN (
		SELECT 
			CASE WHEN (
	            REGEXP_LIKE(A.JBNM, '판사|검사|변호사|법무사|변리사|노무사|회계사|세무사|감정평가사|관세사|의사|치과의사|수의사|한의사|약사|행정사|손해사정사|건축사|도선사|조종사|계리|교수|관제사')
	            OR
	            REGEXP_LIKE(A.JBNM, '전문 간호|항해사|선박 기관사|선장|고위|임원|경무관 이상|소방감 이상|금융 상품|자산|외환 딜러|소프트웨어|프로그래머|데이터|프로그래머|감독|장학관')
	            )
	        AND NOT REGEXP_LIKE(A.JBNM, '장의사|스포츠|건설|코치|검사원|검사공|검사기사|사육|판매|사무원|종사원|의료인|공무원|보조원|현장기술|감독자|제조|공급|사무및감독|경비원')	
			THEN 1 ELSE 0 END AS 전문직
			, CASE WHEN JBNM LIKE '%대표%'
				  AND JBCD LIKE '1%' THEN 1 /* 대표이사 */
				WHEN JBNM LIKE '%임원%' THEN 1
			    WHEN JBNM LIKE '%경영 및 진단 전문가%' THEN 1
			    WHEN JBNM LIKE '%경영인%' THEN 1
			    WHEN JBNM LIKE '%경영부서관리자%' THEN 1
				ELSE 0 END AS 법인대표 
			, CASE WHEN  JBNM LIKE '%자영업자%'      THEN 1
					 WHEN JBNM LIKE '%경영자%' 	      THEN 1

					 WHEN JBNM LIKE '%운영%'		  
					     AND JBNM NOT LIKE '%시스템%'
					     AND JBNM NOT LIKE '%전산%'
					     AND JBNM NOT LIKE '%정보%'
					     AND JBNM NOT LIKE '%기술%'
					     AND JBNM NOT LIKE '%연구%'
					     AND JBNM NOT LIKE '%사무%'
					 THEN 1
					 WHEN JBNM LIKE '%소매%'		  THEN 1
					 WHEN JBNM LIKE '%음식%'		  
					    AND JBNM LIKE '%서비스%' THEN 1
					 WHEN JBNM  LIKE '%조리%'		   
					  AND JBNM NOT LIKE '%산후%'  
					  AND JBNM NOT LIKE '%열차%'
					  AND JBNM NOT LIKE '%선박%'

					  AND JBNM NOT LIKE '%병원%' 				THEN 1
					 WHEN JBNM  LIKE '%피부%'	    			THEN 1
					 WHEN JBNM  LIKE '%커피%'	    			THEN 1
					 WHEN JBNM LIKE '%식당 종사원%' 			THEN 1
					 WHEN JBNM  LIKE '%주방장%'	    			THEN 1
					 WHEN JBNM  LIKE '%미용사%'	    			THEN 1
					 WHEN JBNM  LIKE '%매장 계산원%'	        THEN 1
					 WHEN JBNM  LIKE '%음료서비스원%'	        THEN 1
					 WHEN JBNM  LIKE '%노래방%'	    			THEN 1

					 WHEN JBNM  LIKE '%부동산 컨설턴트%'	    THEN 1

			, CASE WHEN JBNM LIKE '%학생%' THEN 1 ELSE 0 END AS 학생

	AND H.JBCD = B.JBCD
	AND H.JB_CH_SEQNO = B.JB_CH_SEQNO
);

DROP TABLE DATABIZ_FUTVAL_CR_CHNL_PRED;
CREATE TABLE DATABIZ_FUTVAL_CR_CHNL_PRED AS 
SELECT 
	*
FROM 
(SELECT 
	A.CLS_YYMM
	,C.CTMNO
	,A.CHNL
	,COUNT(DISTINCT A.PLYNO) PLYNO_CNT
FROM
(SELECT * FROM SDADAM.DATABIZ_FUTVAL_ALL_CR_TEMP_PRED
WHERE IKD_GRPCD = 'LA') A
INNER JOIN INS_CR_RELPC C
ON 0=0
  	AND A.PLYNO = C.PLYNO
  	AND C.NDS_AP_STR_DTHMS <= F_LAST_DAY(A.CLS_YYMM)
  	AND C.NDS_AP_ND_DTHMS > F_LAST_DAY(A.CLS_YYMM)
  	AND C.RELPC_TPCD IN ('02') /*관계자유형 = 계약자, 피보험자*/
  	AND C.RELPC_STCD IN ('01','04') /*관계자상태코드 = 정상, 납입면제*/
  	AND C.RELPC_DSCNO_FLGCD IN ('01', '05') -- 개인고객(법인고객 제외)
GROUP BY A.CLS_YYMM , C.CTMNO,A.CHNL) 
PIVOT
(
	MAX(PLYNO_CNT)
	FOR CHNL IN ('11' AS 전속,'12' AS 교차,'20' AS GA,'31' AS TM,'32' AS CM,'40' AS 방카,'50' AS 기업,'61' AS LIFEWITH,'62' AS 본사기타)
) P;
	
	
	
DROP TABLE DATABIZ_FUTVAL_CR_CRR_PRED;
CREATE TABLE DATABIZ_FUTVAL_CR_CRR_PRED AS
SELECT 
  CLS_YYMM
  , CTMNO
  , DH_STFNO
  , MAX(INS_ST) AS MX_INS
  , COUNT(DISTINCT CASE WHEN CR_STCD NOT IN ('00' , '12') THEN PLYNO ELSE NULL END) AS CCL_CR_CNT

  (
  SELECT 
  	T.CLS_YYMM
  	, C.CTMNO
  	, A.PLYNO
  	, A.INS_ST
  	, D1.DH_STFNO 
  	, CR_STCD
  FROM DATABIZ_FUTVAL_YYMM_PRED T  
  INNER JOIN INS_INS_CR A

  	AND A.AP_STRDT <= T.CLS_NDDT_TS
  	AND A.AP_NDDT > T.CLS_NDDT_TS
  	AND A.INS_CLSTR > T.CLS_NDDT_TS
  	AND A.INS_ST <= T.CLS_NDDT_TS
  	AND IKD_GRPCD = 'LA'
  INNER JOIN INS_CR_ST_CRR B
  ON 0=0
  	AND A.PLYNO = B.PLYNO
  	AND B.NDS_AP_STR_DTHMS <= T.CLS_NDDT_TS
  	AND B.NDS_AP_ND_DTHMS > T.CLS_NDDT_TS
  INNER JOIN INS_CR_RELPC C
  ON 0=0
  	AND A.PLYNO = C.PLYNO
  	AND C.NDS_AP_STR_DTHMS <= T.CLS_NDDT_TS
  	AND C.NDS_AP_ND_DTHMS > T.CLS_NDDT_TS
  	AND C.RELPC_TPCD IN ('02') /*관계자유형 = 계약자, 피보험자*/
  	AND C.RELPC_STCD IN ('01','04') /*관계자상태코드 = 정상, 납입면제*/
  	AND C.RELPC_DSCNO_FLGCD IN ('01', '05') -- 개인고객(법인고객 제외)
  LEFT JOIN INS_CR_DH_STF_CRR D1
	ON 0=0
		AND A.PLYNO = D1.PLYNO
		AND D1.NDS_AP_STR_DTHMS <= T.CLS_NDDT_TS
		AND D1.NDS_AP_ND_DTHMS > T.CLS_NDDT_TS
		AND D1.DH_STF_TPCD = '01'
		AND D1.PRS_DH_STF_YN = '1'  
  WHERE EXISTS (SELECT 1 FROM DATABIZ_FUTVAL_ALL_CUS_INFO_PRED D WHERE D.CTMNO = C.CTMNO)
)
GROUP BY 
  CLS_YYMM
  , CTMNO
  , DH_STFNO
;

DROP TABLE DATABIZ_FUTVAL_PL_PRED PURGE;
CREATE TABLE DATABIZ_FUTVAL_PL_PRED  AS (	
SELECT *
FROM
(SELECT DISTINCT
	AA.CLS_YYMM --설계년월
	, AA.PLNO --설계번호
	, AA.PLDT --설계일자
	, AA.PL_STCD --설계상태코드
	, AA.PL_ST_CHDT -- 설계상태변경일자
	, AA.DH_STFNO AS CE_STFNO-- 취급직원번호
	, AA.USRNO AS CE_USRNO
	, BB.CTMNO -- 고개번호
	, BB.RELPC_TPCD -- 관계자유형코드
	, AA.PLNM
	, AA.GD_TYPE
	, AA.PL_FIN_CD
  , ROW_NUMBER() OVER (PARTITION BY AA.CLS_YYMM,AA.PLNO ORDER BY AA.PLDT DESC) AS RN
FROM (
	SELECT
		T.CLS_YYMM AS CLS_YYMM --설계년월
		, A.PLNO --설계번호
		, A.PLDT --설계일자
		, A.CGAF_CH_SEQNO -- 발행후변경순번
		, A.PL_STCD -- 설계상태코드
		, A.PL_ST_CHDT -- 설계상태변경일자
		, B.DH_STFNO -- 취급직원번호
		, B.USRNO
		, C.PLNM
		, CASE WHEN E.GDNM LIKE '%Young%' THEN 'L04' -- 자녀
           WHEN E.GDNM LIKE '%더건강 더실속%' OR E.MKTG_GD_CSFCD in ('L02','L01')					THEN 'L01' -- 종합
           WHEN E.GDNM LIKE '%시그니처 여성 건강%'	THEN 'L01'        -- 종합
      	   WHEN E.MKTG_GD_CSFCD = 'L03'							THEN 'L03' -- 운전자
      	   WHEN E.MKTG_GD_CSFCD = 'L04'							THEN 'L04' -- 자녀
      	   WHEN E.MKTG_GD_CSFCD = 'L06'							THEN 'L06' -- 암
      	   WHEN E.MKTG_GD_CSFCD = 'L11'							THEN 'L11' -- 재물
      	   WHEN E.MKTG_GD_CSFCD = 'L08'							THEN 'L08' -- 간병
      	   WHEN E.GDNM LIKE '%세이프투게더%'						THEN 'SAFE' -- 세이프투게더
           WHEN E.GDNM LIKE '%실손의료보험%'						THEN 'SS' -- 실손
      	   WHEN REGEXP_LIKE(E.GDNM, '참 편한|WELL100|간편|3N5|355')  THEN '유병자'
      	   WHEN E.MKTG_GD_CSFCD in ('L12','L13')				THEN 'L123' --연저축가입자수가 부족해서 기타로 분류
      	   ELSE 'ETC' END AS GD_TYPE
       , CASE WHEN A.PL_STCD IN ('03','04','05','06','07','08','09') THEN 1 ELSE 0 END AS PL_FIN_CD  -- 설계완료여부
	FROM DATABIZ_FUTVAL_YYMM_PRED T 
	INNER JOIN INS_INS_PL A
	  ON A.PLDT <= FIRST_DAY(T.CLS_YYMM)
	LEFT OUTER JOIN INS_PL_DH_STF  B    				/* 설계취급자 */
    ON   0=0
		    AND A.PLNO = B.PLNO
        AND A.CGAF_CH_SEQNO = B.CGAF_CH_SEQNO
        AND B.DH_STF_TPCD = '01'                 			/* 모집직원 */
        AND B.IKD_GRPCD = 'LA'
	LEFT JOIN INS_GD_CE_PL C
	ON 0=0
		AND A.PLCD = C.PLCD
		AND A.GDCD = C.GDCD
	LEFT JOIN M_CRM_GD_CSF_INFO E
	ON 0=0
		AND A.GDCD = E.GDCD
	WHERE 1=1
		--AND A.PL_STCD IN ('03','04','05','06','07','08','09') -- '01':설계중 / '02' :단순설계 제외,  '03': 설계완료 ~ '09' : 수납완료
		AND A.PL_FLGCD = '01' -- 설계구분 : "가입"설계
		AND A.IKD_GRPCD = 'LA' -- 장기
		AND A.VALD_PL_YN = '1' -- 유효설계
) AA
INNER JOIN INS_PL_RELPC BB
ON 0=0
	AND AA.PLNO = BB.PLNO
	AND AA.CGAF_CH_SEQNO = BB.CGAF_CH_SEQNO
	AND BB.IKD_GRPCD = 'LA'
	AND BB.RELPC_TPCD IN ('01', '02') -- 계약자/피보험자
WHERE BB.CTMNO IN (SELECT DISTINCT CTMNO FROM DATABIZ_FUTVAL_ALL_CUS_INFO_PRED)
)
WHERE RN = 1)
;

DROP TABLE DATABIZ_FUTVAL_CLM_PRED PURGE;

	,CASE WHEN B.IKD_GRPCD IN ('FA','MA') THEN 'FAMA' ELSE B.IKD_GRPCD END AS IKD_GRPCD

DROP TABLE DATABIZ_FUTVAL_CR_GD_PRED PURGE;
CREATE TABLE DATABIZ_FUTVAL_CR_GD_PRED AS
SELECT 
  *
FROM
  (SELECT 
    CLS_YYMM
    ,CRT_CTMNO
    ,GD_TYPE
    ,COUNT(DISTINCT PLYNO) AS PLY_CNT
  FROM
    (
      SELECT 
        A.*
        , CASE WHEN GDNM LIKE '%Young%' THEN 'L04' -- 자녀
              WHEN GDNM LIKE '%더건강 더실속%' OR MKTG_GD_CSFCD in ('L02','L01')					THEN 'L01' -- 종합
              WHEN GDNM LIKE '%시그니처 여성 건강%'	THEN 'L01'        -- 종합
      	      WHEN MKTG_GD_CSFCD = 'L03'							THEN 'L03' -- 운전자
      	      WHEN MKTG_GD_CSFCD = 'L04'							THEN 'L04' -- 자녀
      	      WHEN MKTG_GD_CSFCD = 'L06'							THEN 'L06' -- 암
      	      WHEN MKTG_GD_CSFCD = 'L11'							THEN 'L11' -- 재물
      	      WHEN MKTG_GD_CSFCD = 'L08'							THEN 'L08' -- 간병
      	      WHEN GDNM LIKE '%세이프투게더%'						THEN 'SAFE' -- 세이프투게더
              WHEN GDNM LIKE '%실손의료보험%'						THEN 'SS' -- 실손
      	      WHEN REGEXP_LIKE(GDNM, '참 편한|WELL100|간편|3N5|355')  THEN '유병자'
      	      WHEN MKTG_GD_CSFCD in ('L12','L13')				THEN 'L123' --연저축가입자수가 부족해서 기타로 분류
      	   	  ELSE 'ETC' END AS GD_TYPE 
      FROM DATABIZ_FUTVAL_ALL_CR_TEMP_PRED A)
  GROUP BY 
    CLS_YYMM
    ,CRT_CTMNO
    ,GD_TYPE)
  PIVOT
  (
  	MAX(PLY_CNT)
  	FOR GD_TYPE IN ('L01' AS L01,'L03' AS L03,'L04' AS L04,'L06' AS L06,'L08' AS L08,'SAFE' AS SAFE , 'SS' AS SS, '유병자'  AS 유병자,'L11' AS L11,'L123' AS L123)
  ) P ;

	, CASE WHEN B13.CTMNO IS NOT NULL THEN 1 ELSE 0 END AS OTHER_CHNL_YN
	, CASE WHEN B15.PSS_CTM_CVDT IS NOT NULL THEN 1 ELSE 0 END AS PSS_CTM_YN
  , CASE WHEN TRUNC(MONTHS_BETWEEN(F_LAST_DAY(A.CLS_YYMM), CASE WHEN TO_CHAR(B15.INP_DTHMS,'YYYY') = '0001' THEN B15.MDF_DTHMS ELSE B15.INP_DTHMS END ),0) >= 60 THEN '5년이상'
        WHEN TRUNC(MONTHS_BETWEEN(F_LAST_DAY(A.CLS_YYMM), CASE WHEN TO_CHAR(B15.INP_DTHMS,'YYYY') = '0001' THEN B15.MDF_DTHMS ELSE B15.INP_DTHMS END ),0) >= 48 THEN '4년이상'
        WHEN TRUNC(MONTHS_BETWEEN(F_LAST_DAY(A.CLS_YYMM), CASE WHEN TO_CHAR(B15.INP_DTHMS,'YYYY') = '0001' THEN B15.MDF_DTHMS ELSE B15.INP_DTHMS END ),0) >= 36 THEN '3년이상'
        WHEN TRUNC(MONTHS_BETWEEN(F_LAST_DAY(A.CLS_YYMM), CASE WHEN TO_CHAR(B15.INP_DTHMS,'YYYY') = '0001' THEN B15.MDF_DTHMS ELSE B15.INP_DTHMS END ),0) >= 24 THEN '2년이상'
        WHEN TRUNC(MONTHS_BETWEEN(F_LAST_DAY(A.CLS_YYMM), CASE WHEN TO_CHAR(B15.INP_DTHMS,'YYYY') = '0001' THEN B15.MDF_DTHMS ELSE B15.INP_DTHMS END ),0) >= 12 THEN '1년이상'

        END AS CUS_INP_DT		
  , CASE WHEN B16.ISPL_AGRE_YN = 1 THEN '동의'
        WHEN B16.ISPL_AGRE_YN = 0 AND TRUNC(MONTHS_BETWEEN(F_LAST_DAY(A.CLS_YYMM),B16.AGRE_NDDT),0) >= 60 THEN '5년이상'
        WHEN B16.ISPL_AGRE_YN = 0 AND TRUNC(MONTHS_BETWEEN(F_LAST_DAY(A.CLS_YYMM),B16.AGRE_NDDT),0) >= 48 THEN '4년이상'

        WHEN B16.ISPL_AGRE_YN = 0 AND TRUNC(MONTHS_BETWEEN(F_LAST_DAY(A.CLS_YYMM),B16.AGRE_NDDT),0) >= 24 THEN '2년이상'
        WHEN B16.ISPL_AGRE_YN = 0 AND TRUNC(MONTHS_BETWEEN(F_LAST_DAY(A.CLS_YYMM),B16.AGRE_NDDT),0) >= 12 THEN '1년이상'
        WHEN B16.ISPL_AGRE_YN = 0 AND TRUNC(MONTHS_BETWEEN(F_LAST_DAY(A.CLS_YYMM),B16.AGRE_NDDT),0) < 12 THEN '1년미만'
        ELSE '동의이력없음'
        END AS ISPL_AGRE	
	, NVL(PRM,0) PRM
  , NVL(B7.L01,0) L01
  , NVL(B7.L03,0) L03
  , NVL(B7.L04,0) L04
  , NVL(B7.L06,0)  L06
  , NVL(B7.L08,0) L08
  , NVL(B7.SAFE,0) SAFE
  , NVL(B7.SS,0) SS
  , NVL(B7.유병자,0) 유병자
  , NVL(B7.L11,0) L11
  , NVL(B7.L123,0) L123
	
FROM
  (
  SELECT *

    AND LA_YN > 0
    AND RELPC_TPCD = '01'

ON 0=0

  AND A.CLS_YYMM = B4.CLS_YYMM

ON 0=0

  AND A.CLS_YYMM = B5.CLS_YYMM

ON 0=0

DROP TABLE CRM_MART_ALL_CUS_INFO PURGE;
CREATE TABLE CRM_MART_ALL_CUS_INFO AS
SELECT 
	A.CLS_YYMM
	,C.CTMNO AS CTMNO  -- 현재 정상 계약의 계약자 OR 피보험자 정보
	,C.RELPC_TPCD AS RELPC_TPCD
	, CASE WHEN F_ISDATE(F_BIRTH_DAY_DEC(C.CTM_DSCNO)) = 'T' THEN TRUNC(MONTHS_BETWEEN(F_LAST_DAY(T.CLS_YYMM), TO_DATE(F_BIRTH_DAY_DEC(C.CTM_DSCNO),'YYYYMMDD')) / 12) 
				  ELSE NULL
				  END AS CTM_AGE
	,CASE WHEN F_SEX_DEC(C.CTM_DSCNO)='M' THEN '1.M'
			WHEN F_SEX_DEC(C.CTM_DSCNO)='F' THEN '2.F'
			ELSE NULL
			END AS CTM_SEX
	,MAX(CASE WHEN C.IKD_GRPCD = 'LA' THEN 1 ELSE 0 END) AS LA_YN
	,MAX(CASE WHEN C.IKD_GRPCD = 'CA' THEN 1 ELSE 0 END) AS CA_YN
	,MAX(CASE WHEN C.IKD_GRPCD IN ('FA','MA') THEN 1 ELSE 0 END) AS FAMA_YN
FROM CRM_MART_ALL_CR_TEMP A
INNER JOIN DATABIZ_FUTVAL_YYMM_PRED T
ON 0=0
	AND A.CLS_YYMM = T.CLS_YYMM
INNER JOIN INS_CR_RELPC C
ON 0=0
	AND A.PLYNO = C.PLYNO
	AND C.NDS_AP_STR_DTHMS <= T.CLS_NDDT
	AND C.NDS_AP_ND_DTHMS > T.CLS_NDDT
	AND C.RELPC_TPCD IN ('01','02') /*관계자유형 = 계약자, 피보험자*/
	AND C.RELPC_STCD IN ('01','04') /*관계자상태코드 = 정상, 납입면제*/
GROUP BY 
	A.CLS_YYMM
	,C.CTMNO   -- 현재 정상 계약의 계약자 OR 피보험자 정보
	,C.RELPC_TPCD 
	, CASE WHEN F_ISDATE(F_BIRTH_DAY_DEC(C.CTM_DSCNO)) = 'T' THEN TRUNC(MONTHS_BETWEEN(F_LAST_DAY(T.CLS_YYMM), TO_DATE(F_BIRTH_DAY_DEC(C.CTM_DSCNO),'YYYYMMDD')) / 12) 
				  ELSE NULL
				  END 
	,CASE WHEN F_SEX_DEC(C.CTM_DSCNO)='M' THEN '1.M'
			WHEN F_SEX_DEC(C.CTM_DSCNO)='F' THEN '2.F'
			ELSE NULL
			END;

        WHEN GDNM LIKE '%시그니처 여성 건강%'	THEN 'L01'        -- 종합

	    WHEN MKTG_GD_CSFCD = 'L04'							THEN 'L04' -- 자녀
	    WHEN MKTG_GD_CSFCD = 'L06'							THEN 'L06' -- 암
	    WHEN MKTG_GD_CSFCD = 'L11'							THEN 'L11' -- 재물
	    WHEN MKTG_GD_CSFCD = 'L08'							THEN 'L08' -- 간병

        WHEN GDNM LIKE '%실손의료보험%'						THEN 'SS' -- 실손

	, B.GN_PRM

	AND A.PLYNO = B.PLYNO
	AND B.AP_STRDT <= F_LAST_DAY(A.CLS_YYMM)
	AND B.AP_NDDT > F_LAST_DAY(A.CLS_YYMM);



DROP TABLE SEG_MkTG_PRES_VAL PURGE;
CREATE TABLE SEG_MKTG_PRES_VAL AS 
SELECT
	DISTINCT
    A.CLS_YYMM
	, A.CTMNO
	, A.CTM_AGE
	, PLY_CNT
	, SM_LTRM_PRM
	, CASE WHEN A.CTM_AGE >= 70 THEN 70
		WHEN A.CTM_AGE >= 60 THEN 60
		WHEN A.CTM_AGE >= 50 THEN 50
		WHEN A.CTM_AGE >= 40 THEN 40
		WHEN A.CTM_AGE >= 30 THEN 30
		WHEN A.CTM_AGE >= 20 THEN 20
		WHEN A.CTM_AGE < 20 THEN 10
		ELSE NULL END AS CTM_AGE_SEG
FROM
(
    SELECT A.* 
    FROM DATABIZ_FUTVAL_YYMM_PRED T
    INNER JOIN SDADAM.CRM_MART_ALL_CUS_INFO A
    ON 0=0
        AND A.CLS_YYMM = T.CLS_YYMM
    WHERE 1=1
    	AND RELPC_TPCD = '02'
    	AND LA_YN = 1
	) A
LEFT JOIN 
	(
    SELECT 
	  A.CLS_YYMM
	  , C.CTMNO 
	  , COUNT(DISTINCT A.PLYNO) PLY_CNT  
	FROM DATABIZ_FUTVAL_YYMM_PRED T
    INNER JOIN SDADAM.CRM_MART_ALL_CR_TEMP A
    ON 0=0
        AND A.CLS_YYMM = T.CLS_YYMM
	INNER JOIN INS_CR_RELPC C
	ON 0=0
		AND A.PLYNO = C.PLYNO
		AND C.NDS_AP_STR_DTHMS <= F_LAST_DAY(A.CLS_YYMM)
		AND C.NDS_AP_ND_DTHMS > F_LAST_DAY(A.CLS_YYMM)
		AND C.RELPC_TPCD IN ('02') /*관계자유형 = 계약자, 피보험자*/
		AND C.RELPC_STCD IN ('01','04') /*관계자상태코드 = 정상, 납입면제*/
	WHERE A.IKD_GRPCD = 'LA'
		AND A.MKTG_GD_CSFCD NOT IN ('L12','L13') 
		AND A.GDNM NOT LIKE '%실손의료보험%'
	GROUP BY 
	  A.CLS_YYMM
	  , C.CTMNO
    ) B
ON 0=0
	AND A.CTMNO = B.CTMNO
    AND A.CLS_YYMM = B.CLS_YYMM
LEFT JOIN
	(
	SELECT 
        A.CLS_YYMM
		, C.CTMNO
		, SUM(A.LTRM_MPY_CV_PRM) AS SM_LTRM_PRM
	FROM DATABIZ_FUTVAL_YYMM_PRED T 
    INNER JOIN CRM_MART_LA_PRM A
    ON 0=0
        AND A.CLS_YYMM = T.CLS_YYMM
	INNER JOIN INS_CR_RELPC C
	ON 0=0
		AND A.PLYNO = C.PLYNO
		AND C.NDS_AP_STR_DTHMS <= F_LAST_DAY(A.CLS_YYMM)

	GROUP BY A.CLS_YYMM, C.CTMNO
	) D
ON 0=0
	AND A.CTMNO = D.CTMNO
    AND A.CLS_YYMM = D.CLS_YYMM

WHERE A.CTMNO NOT IN (SELECT
        	A.CTMNO

DROP TABLE SEG_MKTG_PRES_VAL_T;
CREATE TABLE SEG_MKTG_PRES_VAL_T AS(
    SELECT
        CLS_YYMM
    	,CTMNO
    	,SUM(CASE WHEN 담보명 = '질병사망' THEN 4*SCORE
    			WHEN 담보명 = '질병후유장해' THEN 2*SCORE
    			WHEN 담보명 IN ('간병인사용질병','간병인사용상해') THEN 1.5*SCORE
    			ELSE SCORE END) AS TOTAL_SCORE
    	,MAX(CASE WHEN 담보명 IN ('질병사망','질병후유장해') AND  SCORE > 0 THEN 1 ELSE 0 END) 질병사망
    	,MAX(CASE WHEN 담보명 IN ('유사암진단비','암진단비','뇌혈관질환진단비','허혈성심장질환진단비','급성신근경색증진단비','표적항암약물허가치료비') AND  SCORE > 0 THEN 1 ELSE 0 END) 진단비
    	,MAX(CASE WHEN 담보명 IN ('암수술비','뇌혈관질환수술비','허혈성심장질환수술비','질병수술비','상해수술비') AND  SCORE > 0 THEN 1 ELSE 0 END) 수술비
    	,MAX(CASE WHEN 담보명 IN ('간병인사용질병','간병인사용상해') AND  SCORE > 0 THEN 1 ELSE 0 END) 간병
    FROM 
        (
        SELECT 
            A.CLS_YYMM
            , A.CTMNO
            , A.CTM_AGE_SEG
            , B.담보명
        	, CASE WHEN B.담보명 IN ('표적항암약물허가치료비','간병인사용상해','간병인사용질병') AND 담보가입금액 > 0 THEN 1
        			WHEN B.담보명 IN ('표적항암약물허가치료비','간병인사용상해','간병인사용질병') AND 담보가입금액 = 0 THEN 0
        			WHEN B.담보명 NOT IN ('표적항암약물허가치료비','간병인사용상해','간병인사용질병') AND 담보가입금액 >= HIGH THEN 1
        			WHEN B.담보명 NOT IN ('표적항암약물허가치료비','간병인사용상해','간병인사용질병') AND 담보가입금액 >= LOW THEN 0.75

        			END AS SCORE
        FROM SEG_MKTG_PRES_VAL A
        LEFT JOIN SEG_MKTG_PRES_CVR_INFO B
        ON 0=0
            AND A.CLS_YYMM = B.CLS_YYMM
            AND A.CTMNO = B.CTMNO
        LEFT JOIN SEG_MTKG_CVR_RANGE C

DROP TABLE GDRES_SYS_YYMM_PRED PURGE;
CREATE TABLE GDRES_SYS_YYMM_PRED NOLOGGING AS
SELECT
    YYMM                                                                    AS CLS_YYMM
    , F_ADD_MONTHS_A(F_LAST_DAY(YYMM), -2)                                 AS CLS_YYMM_2MB
    , F_ADD_MONTHS_A(F_LAST_DAY(YYMM), -1)                                 AS CLS_YYMM_1MB
    , FIRST_DAY(YYMM)                                                       AS CLS_STRDT
    , LEAST(SYSDATE, YYMM_NDDT)                                             AS CLS_NDDT
    , LEAST(SYSDATE,
        TO_DATE(TO_CHAR(YYMM_NDDT,'YYYY-MM-DD')||' 23:59:59','YYYY-MM-DD HH24:MI:SS'))
                                                                            AS CLS_NDDT_TS
    , LEAST(CAST(F_LAST_DAY(YYMM) AS DATE), SYSDATE)                       AS CLS_NDDT_LD
FROM M_COD_YYMM
WHERE YYMM = TO_CHAR(SYSDATE, 'YYYYMM');


/* ============================================================================

============================================================================ */
DROP TABLE GDREC_TMGD_LIST PURGE;
CREATE TABLE GDREC_TMGD_LIST NOLOGGING AS
SELECT DISTINCT
    A.*
    , CASE
        WHEN B.GDNM LIKE '%Young%'                                          THEN 'L04'
        WHEN B.GDNM LIKE '%더건강 더실속%' OR B.MKTG_GD_CSFCD IN ('L02','L01') THEN 'L01'
        WHEN B.GDNM LIKE '%더건강%'                                          THEN 'L01'
        WHEN B.GDNM LIKE '%시그니처 여성 건강%'                                THEN 'L01'
        WHEN B.MKTG_GD_CSFCD = 'L03' THEN 'L03'  WHEN B.MKTG_GD_CSFCD = 'L04' THEN 'L04'
        WHEN B.MKTG_GD_CSFCD = 'L06' THEN 'L06'  WHEN B.MKTG_GD_CSFCD = 'L11' THEN 'L11'
        WHEN B.MKTG_GD_CSFCD = 'L08' THEN 'L08'
        WHEN B.GDNM LIKE '%세이프투게더%' THEN 'SAFE'
        WHEN B.GDNM LIKE '%실손의료보험%' THEN 'SS'
        WHEN REGEXP_LIKE(B.GDNM,'참 편한|WELL100|간편|3N5|355') THEN 'SI'
        WHEN B.MKTG_GD_CSFCD IN ('L12','L13') THEN 'L123'
        ELSE 'ETC'
      END AS GD_TYPE
    , B.GDNM
    , D.DTCNM AS SL_CHNCD
FROM IGD_GD_SL_TRM A

/* ============================================================================
   [07] 보유고객 월별 기준 + 보험나이 계산 (T02)
   ※ [성능] INS_AGE 계산 시 동일 AGE() 표현식 2회 반복 → CROSS APPLY로 1회로 축약
   ※ [제거] WHERE A1.IKD_GRPCD='LA' : GDREC_ORIGIN_INS_CR_PRED가 이미 LA 전용
============================================================================ */
DROP TABLE M_BIZ_CTM_INFO_CVR_LIST_RECT_T02 PURGE;
CREATE TABLE M_BIZ_CTM_INFO_CVR_LIST_RECT_T02 NOLOGGING AS
SELECT
    A2.CLS_YYMM, A2.PLYNO, A2.CTMNO, A2.RELPC_SEQNO
    , A2.STFNO, A2.USRNO, A2.GDCD, A2.INS_ST, A2.INS_CLSTR, A2.BRTH
    , CASE
        WHEN F_ISDATE(A2.BRTH) = 'T' AND A2.CTM_TPCD = '01'
        -- CROSS APPLY로 AGE값 1회 계산 후 재사용 (아래 APPLY_AGE 참조)
        THEN A2.AGE_YEAR - CASE
               WHEN A2.AGE_MONTH = 1
                AND SUBSTR(A2.BASE_DT, 6, 2) = SUBSTR(A2.BRTH, 6, 2) THEN 1
               ELSE 0
             END
      END AS INS_AGE
FROM (
    SELECT /*+ PARALLEL(8) FULL(A1) FULL(B1) LEADING(A1) */
        T.CLS_YYMM
        , A1.PLYNO, B1.CTMNO, B1.RELPC_SEQNO
        , A1.DH_STFNO                       AS STFNO
        , COALESCE(A1.DH_USR_NO, '9999999') AS USRNO
        , A1.GDCD, A1.INS_ST, A1.INS_CLSTR
        , F_BIRTH_DAY_DEC(B1.CTM_DSCNO)    AS BRTH
        , C.CTM_TPCD
        -- 기준일(가입월+6개월) 1회 계산
        , TO_CHAR(ADD_MONTHS(F_LAST_DAY(A1.CLS_YYMM), 6)) AS BASE_DT
        -- AGE 결과를 한 번만 계산 → 연/월 추출에 재사용
        , EXTRACT(YEAR FROM
            TO_YMINTERVAL(AGE(ADD_MONTHS(F_LAST_DAY(A1.CLS_YYMM), 6),
                              TO_DATE(F_BIRTH_DAY_DEC(B1.CTM_DSCNO),'YYYYMMDD'))
                          DEFAULT '000-00' ON CONVERSION ERROR)
          ) AS AGE_YEAR
        , EXTRACT(MONTH FROM
            TO_YMINTERVAL(AGE(ADD_MONTHS(F_LAST_DAY(A1.CLS_YYMM), 6),
                              TO_DATE(F_BIRTH_DAY_DEC(B1.CTM_DSCNO),'YYYYMMDD'))
                          DEFAULT '000-00' ON CONVERSION ERROR)
          ) AS AGE_MONTH
        , ROW_NUMBER() OVER (
            PARTITION BY A1.PLYNO, B1.CTMNO, T.CLS_YYMM
            ORDER BY B1.RELPC_SEQNO
          ) AS RN
    FROM (
        SELECT A.*, B.GDNM_CLEAN
        FROM   GDREC_ORIGIN_INS_CR_PRED A
        LEFT JOIN M_DS_CRM_REC_GDNM_TM     B ON A.GDCD = B.GDCD
        WHERE  TO_CHAR(A.INS_ST,'YYYYMM') = F_PREV_MONTH(A.CLS_YYMM,'1')
        -- ※ IKD_GRPCD='LA' 필터 제거: 소스 테이블이 이미 LA 전용
    ) A1
    INNER JOIN GDRES_SYS_YYMM_PRED T  ON A1.CLS_YYMM = T.CLS_YYMM
    INNER JOIN INS_CR_RELPC         B1
        ON  A1.PLYNO            = B1.PLYNO
        AND B1.IKD_GRPCD        IN ('LA')
        AND B1.RELPC_STCD       IN ('01','04')   -- 정상, 납입면제
        AND B1.RELPC_TPCD        = '02'          -- 피보험자
        AND B1.VALD_NDS_YN       = '1'
        AND B1.NDS_AP_STR_DTHMS <= T.CLS_NDDT
        AND B1.NDS_AP_ND_DTHMS   > T.CLS_NDDT
        AND B1.CTMNO            IS NOT NULL
    LEFT JOIN CUS_CTM C ON B1.CTMNO = C.CTMNO
) A2
WHERE A2.RN = 1;

   ※ [성능] EXISTS → INNER JOIN으로 변경 (대용량 INS_CR_CVR 스캔 시 더 효율적)

DROP TABLE M_BIZ_CTM_INFO_CVR_LIST_RECT_T03 PURGE;
CREATE TABLE M_BIZ_CTM_INFO_CVR_LIST_RECT_T03 NOLOGGING AS
SELECT /*+ PARALLEL(8) FULL(A1) FULL(B1) LEADING(A1) */
    A1.CLS_YYMM, A1.PLYNO, B1.CVRCD, B1.CVR_SEQNO
    , A1.CTMNO, A1.RELPC_SEQNO, A1.STFNO, A1.USRNO
    , A1.INS_ST, A1.INS_CLSTR, A1.GDCD, A1.BRTH, A1.INS_AGE
    , B1.CVR_STCD
    , B1.INS_ST                                         AS CVR_INS_ST
    , B1.INS_CLSTR                                      AS CVR_INS_CLSTR
    , F_DIFF_DATE('MM', TO_CHAR(B1.INS_CLSTR,'YYYYMMDD'),
                        TO_CHAR(F_LAST_DAY(A1.CLS_YYMM),'YYYYMMDD')) AS CVR_CLSTR_ST_RNDMC
    , F_DIFF_DATE('MM', TO_CHAR(B1.INS_ST,'YYYYMMDD'),
                        TO_CHAR(F_LAST_DAY(A1.CLS_YYMM),'YYYYMMDD')) AS CVR_ST_ST_RNDMC
    , B1.AP_PRM, B1.ISAMT
    , B1.AUTO_RNW_AV_YN, B1.AUTO_RNW_CVR_CNLDT, B1.RNW_NDDT
    , C1.MDCS_RGT_BJ_YN, C1.RLPMI_CVR_YN
FROM M_BIZ_CTM_INFO_CVR_LIST_RECT_T02 A1
INNER JOIN (
    SELECT
        T.CLS_YYMM, B1.*
        , ROW_NUMBER() OVER (
            PARTITION BY T.CLS_YYMM, B1.PLYNO, B1.RELPC_OJ_SEQNO, B1.CVRCD, B1.CVR_SEQNO
            ORDER BY B1.NDSNO DESC
          ) AS RN
    FROM (
        SELECT
            CVR_STCD, INS_ST, INS_CLSTR, AP_PRM, ISAMT
            , AUTO_RNW_AV_YN, AUTO_RNW_CVR_CNLDT, RNW_NDDT
            , RELPC_OJ_SEQNO, PLYNO, CVRCD, CVR_SEQNO
            , NDS_AP_STR_DTHMS, NDS_AP_ND_DTHMS, NDSNO
        FROM INS_CR_CVR
        -- [성능] EXISTS → INNER JOIN: 옵티마이저가 NL/Hash 선택 가능
        INNER JOIN (SELECT DISTINCT THCP_CVRCD FROM M_BIZ_CTM_INFO_CVR_LIST_RECT_T01) T01
            ON CVRCD = T01.THCP_CVRCD
        WHERE IKD_GRPCD     = 'LA'
          AND VALD_NDS_YN   = '1'
          AND CVR_STCD      IN ('01','07','08')   -- 정상, 완납, 납입면제
          AND CVR_BJ_FLGCD  IN ('01','03')        -- 인담보, 적립담보
    ) B1
    INNER JOIN GDRES_SYS_YYMM_PRED T
        ON  B1.NDS_AP_STR_DTHMS <= T.CLS_NDDT
        AND B1.NDS_AP_ND_DTHMS   > T.CLS_NDDT
) B1
    ON  A1.PLYNO        = B1.PLYNO
    AND A1.RELPC_SEQNO  = B1.RELPC_OJ_SEQNO
    AND A1.CLS_YYMM     = B1.CLS_YYMM
    AND B1.INS_CLSTR   >= F_LAST_DAY(B1.CLS_YYMM)
    AND B1.RN           = 1
LEFT JOIN IGD_GD_CVR C1
    ON  B1.CVRCD    = C1.CVRCD
    AND A1.GDCD     = C1.GDCD
    AND C1.AP_NDDT   > F_LAST_DAY(B1.CLS_YYMM)
    AND C1.AP_STRDT <= F_LAST_DAY(B1.CLS_YYMM);

                WHEN MKTG_GD_CSFCD = 'L03' THEN 'L03'  WHEN MKTG_GD_CSFCD = 'L04' THEN 'L04'
                WHEN MKTG_GD_CSFCD = 'L06' THEN 'L06'  WHEN MKTG_GD_CSFCD = 'L11' THEN 'L11'
                WHEN MKTG_GD_CSFCD = 'L08' THEN 'L08'
                WHEN GDNM LIKE '%세이프투게더%' THEN 'SAFE'
                WHEN GDNM LIKE '%실손의료보험%' THEN 'SS'
                WHEN REGEXP_LIKE(GDNM,'참 편한|WELL100|간편|3N5|355') THEN '유병자'
                WHEN MKTG_GD_CSFCD IN ('L12','L13') THEN 'L123'

   ※ [논리] AP_STRDT/AP_NDDT 기준을 SYSDATE → F_LAST_DAY(CLS_YYMM)으로 변경
      다른 테이블들이 모두 CLS_YYMM 기준이므로 통일 (원본은 SYSDATE → 기준일 불일치)

                  TO_DATE(F_BIRTH_DAY_DEC(B.CTM_DSCNO),'YYYYMMDD')) / 12) < 20

INNER JOIN INS_CR_RELPC B
    ON  A.PLYNO        = B.PLYNO

    AND B.RELPC_TPCD  IN ('02')

/* ============================================================================

DROP TABLE GDREC_ORIGIN_INS_CRTJB_PRED PURGE;
CREATE TABLE GDREC_ORIGIN_INS_CRTJB_PRED NOLOGGING AS
SELECT
    A.*
    , CASE
        WHEN SPJOB_YN = 1             THEN '전문직'
        WHEN STF_YN   = '내근직'      THEN '내근직'

        WHEN JB_TYPE IN ('B1','C','B2') THEN '주부/학생'
        ELSE '블루'
      END AS JB_SEG

                 '장의사|스포츠|건설|코치|검사원|검사공|검사기사|사육|판매|사무원|'||
                 '종사원|의료인|공무원|보조원|현장기술|감독자|제조|공급|사무및감독|경비원')
                THEN 1 ELSE 0
          END AS SPJOB_YN
        , CASE
            WHEN E4.BZ_PART_FLGCD = '09' AND E4.STF_BZ_STCD = '01' THEN '내근직'
            WHEN E4.STFNO IS NOT NULL     AND E4.STF_BZ_STCD = '01' THEN '영업가족'
          END AS STF_YN
        , CASE

            WHEN E1.JBCD='0' AND E1.JB_CH_SEQNO='2'              THEN '1'
            WHEN E1.JBCD='1' AND E1.JB_CH_SEQNO='2'              THEN '2'
            WHEN E1.JBCD='1' AND E1.JB_CH_SEQNO IN ('3','4')     THEN '1'
            WHEN E1.JBCD='2' AND E1.JB_CH_SEQNO IN ('2','3','4') THEN '2'
            WHEN E1.JBCD='3' AND E1.JB_CH_SEQNO='2'              THEN '2'
            WHEN E1.JBCD='3' AND E1.JB_CH_SEQNO IN ('3','4')     THEN '3'
            WHEN E1.JBCD='4' AND E1.JB_CH_SEQNO IN ('3','4')     THEN '4'
            WHEN E1.JBCD='5' AND E1.JB_CH_SEQNO='2'              THEN '4'
            WHEN E1.JBCD IN ('6','8','9') AND E1.JB_CH_SEQNO='1' THEN '4'
            WHEN E1.JBCD='4' AND E1.JB_CH_SEQNO='2'              THEN '5'
            WHEN E1.JBCD='5' AND E1.JB_CH_SEQNO IN ('3','4')     THEN '5'
            WHEN E1.JBCD='1' AND E1.JB_CH_SEQNO='1'              THEN '6'
            WHEN E1.JBCD='6' AND E1.JB_CH_SEQNO IN ('3','4')     THEN '6'
            WHEN E1.JBCD='9' AND E1.JB_CH_SEQNO='2'              THEN '6'
            WHEN E1.JBCD='4' AND E1.JB_CH_SEQNO='1'              THEN '7'
            WHEN E1.JBCD='5' AND E1.JB_CH_SEQNO='1'              THEN '7'
            WHEN E1.JBCD IN ('6','7','8') AND E1.JB_CH_SEQNO='2' THEN '7'
            WHEN E1.JBCD='7' AND E1.JB_CH_SEQNO IN ('3','4')     THEN '7'
            WHEN E1.JBCD IN ('3','7') AND E1.JB_CH_SEQNO='1'     THEN '8'
            WHEN E1.JBCD='8' AND E1.JB_CH_SEQNO IN ('3','4')     THEN '8'

            AND BB.AP_STR_DTHMS <= F_LAST_DAY(AA.CLS_YYMM)
            AND BB.AP_ND_DTHMS   > F_LAST_DAY(AA.CLS_YYMM)
    ) A
    LEFT JOIN CUS_JB        E1 ON SUBSTR(A.JBCD,1,1) = E1.JBCD AND A.JB_CH_SEQNO = E1.JB_CH_SEQNO
    LEFT JOIN CUS_JB        E2 ON A.JBCD = E2.JBCD AND A.JB_CH_SEQNO = E2.JB_CH_SEQNO
    LEFT JOIN CUS_CTM       E3 ON A.CTMNO = E3.CTMNO
    LEFT JOIN M_ORG_BZ_ORGN E4 ON E3.CTM_DSCNO = E4.RSNO

/* ============================================================================
   [20] 지급 보험금 (장기 + 자동차)
   ※ [성능] LEFT JOIN + WHERE B.컬럼 → 사실상 INNER JOIN. 명시적으로 변경.
   ※ [성능] 원본 UNION → UNION ALL: 두 소스는 완전히 다른 계통
============================================================================ */
DROP TABLE GDREC_ORIGIN_INS_CLM_PRED PURGE;
CREATE TABLE GDREC_ORIGIN_INS_CLM_PRED NOLOGGING AS
-- 장기 보험
SELECT
    T.CLS_YYMM, T.CLS_NDDT, A.MN_NRDPS_CTMNO
    , CASE WHEN B.IKD_GRPCD IN ('FA','MA') THEN 'FAMA' ELSE B.IKD_GRPCD END AS IKD_GRPCD
    , B.CLMDT
    , SUM(B.WONCR_CV_PYAMT) AS PY_AMT
FROM GDRES_SYS_YYMM_PRED T
INNER JOIN GDREC_ORIGIN_INS_CR_PRED    A ON T.CLS_YYMM = A.CLS_YYMM
INNER JOIN M_CMP_GN_CVRCL_PY_IBNF_LIST B            -- [수정] LEFT→INNER (WHERE로 필터 중)
    ON  A.MN_NRDPS_CTMNO = B.CLMPS_CTMNO
    AND B.CLMDT BETWEEN F_LAST_DAY(F_PREV_MONTH(T.CLS_YYMM,'60')) AND T.CLS_NDDT
GROUP BY T.CLS_YYMM, T.CLS_NDDT, A.MN_NRDPS_CTMNO
       , CASE WHEN B.IKD_GRPCD IN ('FA','MA') THEN 'FAMA' ELSE B.IKD_GRPCD END, B.CLMDT

UNION ALL  -- [수정] UNION → UNION ALL: 소스가 다른 테이블이므로 중복 없음

-- 자동차 보험
SELECT CLS_YYMM, CLS_NDDT, MN_NRDPS_CTMNO, IKD_GRPCD, CLMDT
    , SUM(RPBL_PY_IBAMT) + SUM(OPTN_PY_IBAMT) AS PY_AMT
FROM (
    SELECT DISTINCT
        T.CLS_YYMM, T.CLS_NDDT, A.MN_NRDPS_CTMNO, 'CA' AS IKD_GRPCD
        , C.CLMDT, B.RPBL_PY_IBAMT, B.OPTN_PY_IBAMT
    FROM GDRES_SYS_YYMM_PRED T
    INNER JOIN GDREC_ORIGIN_INS_CR_PRED A ON T.CLS_YYMM = A.CLS_YYMM
    LEFT  JOIN M_CMP_CR_CLM_CRST        B ON A.MN_NRDPS_CTMNO = B.NRDPS_CTMNO
    INNER JOIN M_CMP_CR_CMP_CLM_CR      C            -- [수정] LEFT→INNER
        ON  B.RCPDT = C.RCPDT AND B.RCP_NV_SEQNO = C.RCP_NV_SEQNO
        AND C.CLMDT BETWEEN F_LAST_DAY(F_PREV_MONTH(T.CLS_YYMM,'60')) AND T.CLS_NDDT
) AA
GROUP BY CLS_YYMM, CLS_NDDT, MN_NRDPS_CTMNO, IKD_GRPCD, CLMDT;

        WHEN E.GDNM LIKE '%더건강%'                                          THEN 'L01'
        WHEN E.GDNM LIKE '%시그니처 여성 건강%'                                THEN 'L01'
        WHEN E.MKTG_GD_CSFCD = 'L03' THEN 'L03'  WHEN E.MKTG_GD_CSFCD = 'L04' THEN 'L04'
        WHEN E.MKTG_GD_CSFCD = 'L06' THEN 'L06'  WHEN E.MKTG_GD_CSFCD = 'L11' THEN 'L11'

        WHEN REGEXP_LIKE(E.GDNM,'참 편한|WELL100|간편|3N5|355') THEN '유병자'
        WHEN E.MKTG_GD_CSFCD IN ('L12','L13') THEN 'L123'
        ELSE 'ETC'
      END AS GD_TYPE
    , C.INS_ST
FROM (SELECT DISTINCT CRT_CTMNO, MN_NRDPS_CTMNO, CLS_YYMM FROM GDREC_ORIGIN_INS_CR_PRED) A

    ON  B.PLYNO    = C.PLYNO
    AND C.AP_STRDT <= F_LAST_DAY(A.CLS_YYMM) AND C.AP_NDDT > F_LAST_DAY(A.CLS_YYMM)
    AND C.IKD_GRPCD = 'LA'
INNER JOIN INS_CR_ST_CRR D      -- CR_STCD 조회용 (NDS 기준)
    ON  D.PLYNO              = B.PLYNO
    AND D.NDS_AP_STR_DTHMS  <= F_LAST_DAY(A.CLS_YYMM)
    AND D.NDS_AP_ND_DTHMS    > F_LAST_DAY(A.CLS_YYMM)
INNER JOIN M_CRM_GD_CSF_INFO E ON C.GDCD = E.GDCD
WHERE D.CR_STCD NOT IN ('00','10');

LEFT JOIN GDREC_ORIGIN_INS_CRTJB_PRED  B4  ON A.CLS_YYMM = B4.CLS_YYMM AND A.CRT_CTMNO = B4.CTMNO
LEFT JOIN (
    SELECT CLS_YYMM, MN_NRDPS_CTMNO
        , COUNT(DISTINCT CLMDT) AS CLM_CNT
        , SUM(PY_AMT)           AS PY_AMT
    FROM GDREC_ORIGIN_INS_CLM_PRED
    GROUP BY CLS_YYMM, MN_NRDPS_CTMNO
) B5 ON A.CLS_YYMM = B5.CLS_YYMM AND A.MN_NRDPS_CTMNO = B5.MN_NRDPS_CTMNO
LEFT JOIN (
    SELECT CLS_YYMM, CTMNO
        , MAX(NO_CVR_YN) AS NO_CVR_YN, MAX(XC_YN) AS XC_YN
    FROM GDREC_ORIGIN_INS_NOCVR_XC_PRED
    GROUP BY CLS_YYMM, CTMNO
) B6 ON A.CLS_YYMM = B6.CLS_YYMM AND A.MN_NRDPS_CTMNO = B6.CTMNO
LEFT JOIN GDREC_ORIGIN_INS_CRTINFO_PRED  B7  ON A.CLS_YYMM = B7.CLS_YYMM  AND A.CRT_CTMNO      = B7.CRT_CTMNO
LEFT JOIN GDREC_ORIGIN_INS_NRDINFO_PRED  B77 ON A.CLS_YYMM = B77.CLS_YYMM AND A.MN_NRDPS_CTMNO = B77.MN_NRDPS_CTMNO
LEFT JOIN (
    SELECT CLS_YYMM, CTMNO
        , SUM(CASE WHEN DU_AR_FLGCD='01' THEN AP_PRM END) AS AP_PRM
        , SUM(CASE WHEN DU_AR_FLGCD='02' THEN AP_PRM END) AS DELAY_PRM
        , COUNT(DISTINCT CASE WHEN DU_AR_FLGCD='02' THEN CLS_YYMM END) AS DELAY_YYMM
        , SUM(CASE WHEN DU_AR_FLGCD='02' AND NW_AR_CTUAR_FLGCD='03' THEN AP_PRM END) AS CTU_DELAY_PRM
        , COUNT(DISTINCT CASE WHEN DU_AR_FLGCD='02' AND NW_AR_CTUAR_FLGCD='03' THEN CLS_YYMM END) AS CTU_DELAY_YYMM
    FROM GDREC_ORIGIN_INS_OVD_PRED
    GROUP BY CLS_YYMM, CTMNO
) B8 ON A.CLS_YYMM = B8.CLS_YYMM AND A.CRT_CTMNO = B8.CTMNO;


/* ============================================================================
   [24] 설계 정보
   ※ [성능] INNER JOIN BBB WHERE CTMNO IN (SELECT DISTINCT ...) →
      EXISTS로 변경 (PRMCNT가 이미 유니크키 수준이므로 semi-join이 더 효율적)
============================================================================ */
DROP TABLE GDREC_ORIGIN_PL_PRED PURGE;
CREATE TABLE GDREC_ORIGIN_PL_PRED NOLOGGING AS
SELECT *
FROM (
    SELECT DISTINCT
        AA.PL_YYMM, AA.PLNO, AA.PLDT, AA.PL_STCD, AA.PL_ST_CHDT
        , AA.DH_STFNO AS CE_STFNO, AA.USRNO AS CE_USRNO
        , BB.CTMNO AS CRT_CTMNO, BBB.CTMNO AS MN_NRDPS_CTMNO
        , AA.PLNM, AA.GD_TYPE, AA.PL_FIN_CD
        , ROW_NUMBER() OVER (PARTITION BY AA.PLNO ORDER BY AA.PLDT DESC) AS RN
    FROM (
        SELECT
            TO_CHAR(A.PLDT,'YYYYMM') AS PL_YYMM, A.PLNO, A.PLDT, A.CGAF_CH_SEQNO
            , A.PL_STCD, A.PL_ST_CHDT, B.DH_STFNO, B.USRNO, C.PLNM
            , CASE
                WHEN E.GDNM LIKE '%Young%'                                          THEN 'L04'
                WHEN E.GDNM LIKE '%더건강 더실속%' OR E.MKTG_GD_CSFCD IN ('L02','L01') THEN 'L01'
                WHEN E.GDNM LIKE '%더건강%'                                          THEN 'L01'
                WHEN E.GDNM LIKE '%시그니처 여성 건강%'                                THEN 'L01'
                WHEN E.MKTG_GD_CSFCD = 'L03' THEN 'L03'  WHEN E.MKTG_GD_CSFCD = 'L04' THEN 'L04'
                WHEN E.MKTG_GD_CSFCD = 'L06' THEN 'L06'  WHEN E.MKTG_GD_CSFCD = 'L11' THEN 'L11'
                WHEN E.MKTG_GD_CSFCD = 'L08' THEN 'L08'
                WHEN E.GDNM LIKE '%세이프투게더%' THEN 'SAFE'
                WHEN E.GDNM LIKE '%실손의료보험%' THEN 'SS'
                WHEN REGEXP_LIKE(E.GDNM,'참 편한|WELL100|간편|3N5|355') THEN '유병자'
                WHEN E.MKTG_GD_CSFCD IN ('L12','L13') THEN 'L123'
                ELSE 'ETC'
              END AS GD_TYPE
            , CASE WHEN A.PL_STCD IN ('03','04','05','06','07','08','09') THEN 1 ELSE 0 END AS PL_FIN_CD
        FROM INS_INS_PL A
        LEFT JOIN INS_PL_DH_STF  B ON A.PLNO=B.PLNO AND A.CGAF_CH_SEQNO=B.CGAF_CH_SEQNO
                                  AND B.DH_STF_TPCD='01' AND B.IKD_GRPCD='LA'
        LEFT JOIN INS_GD_CE_PL   C ON A.PLCD=C.PLCD AND A.GDCD=C.GDCD
        LEFT JOIN M_CRM_GD_CSF_INFO E ON A.GDCD=E.GDCD
        WHERE A.PL_FLGCD='01' AND A.IKD_GRPCD='LA' AND A.VALD_PL_YN='1'
    ) AA
    INNER JOIN INS_PL_RELPC BB
        ON  AA.PLNO=BB.PLNO AND AA.CGAF_CH_SEQNO=BB.CGAF_CH_SEQNO
        AND BB.IKD_GRPCD='LA' AND BB.RELPC_TPCD IN ('01')
        -- [성능] IN(SELECT DISTINCT ...) → EXISTS (semi-join, PRMCNT 재스캔 방지)
        AND EXISTS (SELECT 1 FROM GDREC_ORIGIN_INS_PRMCNT_PRED P WHERE BB.CTMNO = P.CRT_CTMNO)
    INNER JOIN INS_PL_RELPC BBB
        ON  AA.PLNO=BBB.PLNO AND AA.CGAF_CH_SEQNO=BBB.CGAF_CH_SEQNO
        AND BBB.IKD_GRPCD='LA' AND BBB.RELPC_TPCD IN ('02')
        AND EXISTS (SELECT 1 FROM GDREC_ORIGIN_INS_PRMCNT_PRED P WHERE BBB.CTMNO = P.MN_NRDPS_CTMNO)
)
WHERE RN = 1;


/* ============================================================================
   [25] 설계 이력 건수 PIVOT
   [26] 설계 완료 건수 PIVOT
   ※ 두 테이블은 WHERE PL_FIN_CD=1 유무만 다름 → 공통 인라인 뷰 구조 사용
============================================================================ */
DROP TABLE GDREC_ORIGIN_PL_ALL_PRED PURGE;
CREATE TABLE GDREC_ORIGIN_PL_ALL_PRED NOLOGGING AS
SELECT *
FROM (
    SELECT CLS_YYMM, CRT_CTMNO, MN_NRDPS_CTMNO, GD_TYPE
        , COUNT(DISTINCT PLNO) AS PLNO_CNT
    FROM (
        SELECT A.*, B.PLDT, B.GD_TYPE, B.PLNO
            , RANK() OVER (PARTITION BY A.CRT_CTMNO, A.MN_NRDPS_CTMNO ORDER BY A.CLS_YYMM DESC) AS RN
        FROM GDREC_ORIGIN_INS_PRMCNT_PRED A
        LEFT JOIN GDREC_ORIGIN_PL_PRED    B
            ON  A.CRT_CTMNO = B.CRT_CTMNO AND A.MN_NRDPS_CTMNO = B.MN_NRDPS_CTMNO
            AND F_LAST_DAY(A.CLS_YYMM) >= B.PLDT
    )
    WHERE RN = 1
    GROUP BY CLS_YYMM, CRT_CTMNO, MN_NRDPS_CTMNO, GD_TYPE
)
PIVOT (MAX(PLNO_CNT) FOR GD_TYPE IN (
    'L01' AS L01,'L03' AS L03,'L04' AS L04,'L06' AS L06,'L08' AS L08,
    'SAFE' AS SAFE,'SS' AS SS,'유병자' AS 유병자,'L11' AS L11,'L123' AS L123
)) P;

CREATE TABLE GDREC_ORIGIN_PL_CL_PRED NOLOGGING AS
SELECT *
FROM (

            , RANK() OVER (PARTITION BY A.CRT_CTMNO, A.MN_NRDPS_CTMNO ORDER BY A.CLS_YYMM DESC) AS RN
        FROM GDREC_ORIGIN_INS_PRMCNT_PRED A
        LEFT JOIN GDREC_ORIGIN_PL_PRED    B
            ON  A.CRT_CTMNO = B.CRT_CTMNO AND A.MN_NRDPS_CTMNO = B.MN_NRDPS_CTMNO
            AND F_LAST_DAY(A.CLS_YYMM) >= B.PLDT
        WHERE B.PL_FIN_CD = 1          -- 설계 완료만
    )
    WHERE RN = 1
    GROUP BY CLS_YYMM, CRT_CTMNO, MN_NRDPS_CTMNO, GD_TYPE

PIVOT (MAX(PLNO_CNT) FOR GD_TYPE IN (
    'L01' AS L01,'L03' AS L03,'L04' AS L04,'L06' AS L06,'L08' AS L08,
    'SAFE' AS SAFE,'SS' AS SS,'유병자' AS 유병자,'L11' AS L11,'L123' AS L123
)) P;

DROP TABLE GDREC_ORIGIN_MAIN_FIN_PRED PURGE;
CREATE TABLE GDREC_ORIGIN_MAIN_FIN_PRED NOLOGGING AS
SELECT
    A.*
    , B1.L01_설계이력, B1.L03_설계이력, B1.L04_설계이력, B1.L06_설계이력, B1.L08_설계이력
    , B1.SAFE_설계이력, B1.SS_설계이력, B1.유병자_설계이력, B1.L11_설계이력, B1.L123_설계이력
    , B1.L01_설계완료, B1.L03_설계완료, B1.L04_설계완료, B1.L06_설계완료, B1.L08_설계완료
    , B1.SAFE_설계완료, B1.SS_설계완료, B1.유병자_설계완료, B1.L11_설계완료, B1.L123_설계완료
    , B1.최근가입설계상품
    -- NULL='X', >10='10', 그 외=연수 문자열
    , CASE WHEN B2.입원청구    > 10 THEN '10' WHEN B2.입원청구    IS NULL THEN 'X' ELSE TO_CHAR(B2.입원청구)    END AS 입원청구
    , CASE WHEN B2.입원고지    > 10 THEN '10' WHEN B2.입원고지    IS NULL THEN 'X' ELSE TO_CHAR(B2.입원고지)    END AS 입원고지
    , CASE WHEN B2.중대질환고지 > 10 THEN '10' WHEN B2.중대질환고지 IS NULL THEN 'X' ELSE TO_CHAR(B2.중대질환고지) END AS 중대질환고지
    , CASE WHEN B2.중대질환청구 > 10 THEN '10' WHEN B2.중대질환청구 IS NULL THEN 'X' ELSE TO_CHAR(B2.중대질환청구) END AS 중대질환청구
    , CASE WHEN B2.무사고      > 10 THEN '10' WHEN B2.무사고      IS NULL THEN 'X' ELSE TO_CHAR(B2.무사고)      END AS 무사고
FROM GDREC_ORIGIN_MAIN9_PRED A
LEFT JOIN GDREC_TMGD_PL_INFO_PRED  B1
    ON A.CRT_CTMNO=B1.CRT_CTMNO AND A.CLS_YYMM=B1.CLS_YYMM AND A.MN_NRDPS_CTMNO=B1.MN_NRDPS_CTMNO
LEFT JOIN GDREC_TMGD_CLM_INFO_PRED B2
    ON A.CRT_CTMNO=B2.CRT_CTMNO AND A.CLS_YYMM=B2.CLS_YYMM AND A.MN_NRDPS_CTMNO=B2.MN_NRDPS_CTMNO;
