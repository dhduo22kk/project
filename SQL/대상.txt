/* 기준연월 SETTING */
DROP TABLE TBL_DS_YYMM;
CREATE TABLE TBL_DS_YYMM AS (
            SELECT
                YYMM AS CLS_YYMM
                ,F_ADD_MONTHS_A(F_LAST_DAY(YYMM),-6) AS CLS_YYMM_6MB
                ,F_ADD_MONTHS_A(F_LAST_DAY(YYMM),-2) AS CLS_YYMM_2MB
                ,F_ADD_MONTHS_A(F_LAST_DAY(YYMM),-1) AS CLS_YYMM_1MB
                ,FIRST_DAY(YYMM) AS CLS_STRDT
                ,LEAST(SYSDATE, YYMM_NDDT) AS CLS_NDDT
                ,LEAST(SYSDATE, TO_DATE(TO_CHAR(YYMM_NDDT, 'YYYY-MM-DD') || ' 23:59:59', 'YYYY-MM-DD HH24:MI:SS')) AS CLS_NDDT_TS
                ,LEAST(CAST(F_LAST_DAY(YYMM) AS DATE), SYSDATE) AS CLS_NDDT_LD
            FROM M_COD_YYMM
            WHERE YYMM = TO_CHAR(SYSDATE,'YYYYMM')
            );

/*1-1. 장기 TM 대상자 보유 고객 추출 쿼리 - 장기 TM의 계약을 보유한 고객이면 대상 */
DROP TABLE TBL_DS_CRLIST PURGE;
CREATE TABLE TBL_DS_CRLIST NOLOGGING AS(
SELECT
  	LCA.CLS_YYMM
  	, 'LA'			AS IKD_GRPCD
  	, LCA.PLYNO
  	, LCA.CNRDT
  	, LCA.CRT_CTMNO	AS CTMNO
  	, LCA.DH_STFNO
  FROM HWMRT.M_LCR_MTHY_LTINS_CR LCA
  INNER JOIN SDADAM.TBL_DS_YYMM T ON T.CLS_YYMM = LCA.CLS_YYMM
  INNER JOIN M_BIZ_MTHY_PS_CR PCA
  ON 0=0
    AND 	LCA.CLS_YYMM	= PCA.CLS_YYMM
    AND 	LCA.PLYNO		= PCA.PLYNO
  WHERE 1=1
  	AND LCA.DH_STFNO IN (SELECT STFNO FROM M_ORG_BZ_ORGN WHERE HDQT_ORGCD = '1300012' AND BRNM LIKE '%장기%' AND STF_FLGCD = '02'));


DROP TABLE TBL_DS_CUSLIST PURGE;
CREATE TABLE TBL_DS_CUSLIST NOLOGGING AS(
SELECT
DISTINCT 
    LCB.CLS_YYMM
    , LCB.IKD_GRPCD
    , LCB.PLYNO
    , LCB.CNRDT
    , LCB.CTMNO
    , LCB.DH_STFNO
    , LPA.CMS_CTMNO
    , MCA.PRS_TM_CTMNO

	  , MCA.MKTG_TL_RCV_YN										-- 마케팅전화수신여부

ON 0=0
	AND LCB.CLS_YYMM = MCA.CLS_YYMM
	AND LPA.CMS_CTMNO = MCA.CMS_CTMNO

  AND PCA.PLYNO IS NOT NULL)

/*1-2. 장기 TM 대상자 가망 고객 추출 쿼리 - TM 마동 고객이거나 1년내 캠페인 배정 이력 있으면 가망고객  */
DROP TABLE TBL_DS_PSSLIST PURGE;
CREATE TABLE TBL_DS_PSSLIST NOLOGGING AS (
SELECT DISTINCT *
FROM
    (
         SELECT                                   -- 2. 마케팅동의(전화)가 유효하고, TB_CUSTOMER 등록 가망고객
           B.CLS_YYMM
           , B.INP_DTHMS
           , A.CMS_CTMNO
           , A.PRS_CTMNO AS CTMNO
           , A.PRS_TM_CTMNO TM_CUSTID
           , B.INP_USR_ID  
           , '비대면등록_CUSTID' AS INP_SEG
      	   , A.FNL_MKTG_AGRYN										-- 최종마케팅동의여부
      	   , A.MKTG_TL_RCV_YN										-- 마케팅전화수신여부
      	   , A.FNL_MKTG_AGRE_NDDT                -- 마케팅동의종료일자
        FROM HWMRT.M_CMS_CTMCS A
        INNER JOIN 
        	(SELECT	
        		T.CLS_YYMM
        		, A.*
        	FROM SDADAM.TBL_DS_YYMM T
        	INNER JOIN TB_CUSTOMER A
        	ON 0=0
        		AND A.INP_DTHMS <= T.CLS_NDDT_TS
        	WHERE INP_USR_ID IN (SELECT STFNO FROM M_ORG_BZ_ORGN WHERE HDQT_ORGCD = '1300012')
        		) B
        ON 1=1
        	AND A.PRS_TM_CTMNO = B.CUSTID        
    	    AND A.FNL_MKTG_AGRYN = 1
    	    AND A.MKTG_TL_RCV_YN = 1
    
    UNION ALL                               

        SELECT                                   -- 3. 마케팅동의(전화)가 유효하고, 직전1년이내 캠페인으로 배정
        	DISTINCT                           
        	  T.CLS_YYMM
        	  , X2.ASDT AS INP_DTHMS
        	  , X2.CMS_CTMNO
            , X2.PRS_CTMNO AS CTMNO
            , X2.PRS_TM_CTMNO TM_CUSTID
            , CMPG_ALLCT_STFNO AS INP_USR_ID
        	  , '캠페인배정' AS INP_SEG
      	    , X3.FNL_MKTG_AGRYN										-- 최종마케팅동의여부
      	    , X3.MKTG_TL_RCV_YN										-- 마케팅전화수신여부
      	    , X3.FNL_MKTG_AGRE_NDDT                -- 마케팅동의종료일자
        FROM SDADAM.TBL_DS_YYMM T
        INNER JOIN HWMRT.M_CMS_CTM_CLS X1 /*CMS고객마감*/
        ON T.CLS_YYMM = X1.CLS_YYMM
        INNER JOIN HWMRT.M_CMS_INR_CHN_CMPG_DB_CLS X2 /*CMS캠페인DB현황*/
        ON X1.CMS_CTMNO = X2.CMS_CTMNO /*CMS고객번호*/
        AND X1.CLS_YYMM = X2.CLS_YYMM
        AND X2.ASDT >= CLS_NDDT_TS - INTERVAL '1' YEAR 
        INNER JOIN HWMRT.M_CMS_CTMCS X3
        ON X1.CMS_CTMNO = X3.CMS_CTMNO
        AND X3.FNL_MKTG_AGRYN = 1
        AND X3.MKTG_TL_RCV_YN = 1
        WHERE CMPG_LCLCD IN ('C')
        )
    WHERE CMS_CTMNO NOT IN (SELECT DISTINCT CMS_CTMNO FROM SDADAM.TBL_DS_CUSLIST)
);

SELECT
    	DISTINCT
    	 B.TASKID 
    	, B3.TASKNAME 
    	, B.LISTGROUPID 
    	, B4.LISTNAME     리스트그룹
    	, B.LISTID  
    	, B5.LISTNAME     리스트

    	, C.CMS_CMPG_NM

    ON C.LIST_NO = B.LISTID
      AND C.TASK_NO = B.TASKID
      AND C.LIST_GPNO = B.LISTGROUPID
    WHERE 1=1
      AND NVL(B.DELYN,'N')        <> 'Y'
      AND B.ALLOCCHARGERID IS NOT NULL
      AND B.CUSTID IN (SELECT DISTINCT PRS_TM_CTMNO FROM TBL_DS_CTM_LIST);

/***************************************************************************/
/* 전화안내방법동의여부 (상품소개동의)                                     */
/* 추출 출처 : 태동일C, 자동차POM_DB                                       */
/***************************************************************************/
--EXEC P_DROP_TABLE ('E_TM_LTRM_POM_EXTR_KDH_T12');
DROP TABLE E_TM_LTRM_POM_EXTR_KDH_T12;
CREATE TABLE E_TM_LTRM_POM_EXTR_KDH_T12
AS
SELECT  A2.CTM_DSCNO                                     /* 고객식별번호   */
      , A2.TL_NTC_MTD_AGRYN                              /* 전화안내방법동의여부 */
      , TRUNC(A2.AP_STR_DTHMS)  AS TL_NTC_MTD_AGR_STR_DT /* 전화안내방법도의시작일자 */
      , TRUNC(A2.AP_ND_DTHMS)   AS TL_NTC_MTD_AGR_ND_DT  /* 전화안내방법도의종료일자 */
FROM (
       SELECT  B2.CTM_DSCNO          AS CTM_DSCNO              /* 고객식별번호   */
             , CASE WHEN A1.INP_DTHMS    >=  TRUNC(F_ADD_MONTHS_GP(CURRENT_DATE, -24))                        /* 금감원기준 2년까지만 상품소개동의가능 */
                    AND  A1.AP_ND_DTHMS  >   TRUNC(CURRENT_DATE)                                              /* 적용종료일시 */
                    AND  A1.AP_STR_DTHMS <=  CAST(TO_CHAR(CURRENT_DATE,'YYYYMMDD')||'235959' AS TIMESTAMP(0)) /* 적용시작일시 */
               THEN '1'
               ELSE '0'
               END                   AS TL_NTC_MTD_AGRYN     /* 전화안내방법동의여부 */
             , A1.AP_STR_DTHMS       AS AP_STR_DTHMS         /* 적용시작일시         */
             , A1.AP_ND_DTHMS        AS AP_ND_DTHMS          /* 적용종료일시         */
             , ROW_NUMBER() OVER (PARTITION BY B2.CTM_DSCNO
                                  ORDER BY     CASE WHEN  A1.INP_DTHMS           >=  TRUNC(F_ADD_MONTHS_GP(CURRENT_DATE, -24))                                     /* 금감원기준 2년까지만 상품소개동의가능 */
                                                    AND   A1.AP_ND_DTHMS         >   TRUNC(CURRENT_DATE)                                                           /* 적용종료일시 */
                                                    AND   A1.AP_STR_DTHMS        <=  CAST(TO_CHAR(CURRENT_DATE,'YYYYMMDD')||'235959' AS TIMESTAMP(0))     /* 적용시작일시 */
                                               THEN '1'
                                               ELSE '0'
                                               END             DESC  /* 전화안내방법동의여부 */
                                             , A1.AP_ND_DTHMS DESC
                                             , A1.AP_STR_DTHMS DESC)   AS RN
       FROM (
             SELECT  DISTINCT
                     B1.CTM_DSCNO
                   , C1.CUSTID
             FROM  E_TM_LTRM_POM_EXTR_KDH_T01 B1
             INNER JOIN TB_CUSTOMER C1              /** 고객마스터 **/
             ON    C1.CUST_DSCNO = B1.CTM_DSCNO
            ) B2
       INNER JOIN TB_CTM_CRDIU_AGRE A1              /** 고객신용정보활용동의 **/
       ON    A1.CUSTID              = B2.CUSTID
       WHERE 1=1
       AND   A1.CRDIU_AGRE_FLGCD     = '03'               /* 신용정보활용동의구분코드  03:상품소개 */
       AND   A1.CRDIU_AGRE_DTL_FLGCD IN( '01','02','03')  /* 신용정보활용동의세부구분코드 01:처리이용,02:조회이용,03:제공이용 */
     ) A2
WHERE RN = 1
;

/*캠페인 팀별 구분 */
SELECT * 
FROM
(SELECT 
	A.CLS_YYMM 
	, CASE WHEN TMNM LIKE '%POM%' THEN '[C02]POM'
		WHEN TMNM LIKE '%웹인%' THEN '[C01]웹인'
		WHEN TMNM LIKE '%미디어%' THEN '[C14]미디어'
		WHEN TMNM LIKE '%O/B%' THEN '[C07]O/B'
		WHEN TMNM LIKE '%혼합%' THEN '[C16]혼합'
		WHEN TMNM LIKE '%직인%' THEN '[C03]직인'
		WHEN TMNM LIKE '%호전환%' THEN '[C09]호전환'
		WHEN TMNM LIKE '%채팅%' THEN '[C08]채팅'
		WHEN TMNM LIKE '%인콜%' THEN '[C18]인콜'
		ELSE NULL END CMPG_MDCCD
	, SUM(CASE WHEN CMPG_LCLCD = 'B' THEN CR_CRCC_RNPRM
			ELSE  LTRM_CRCC_MPY_CVPRM END) AS CR_PRM
FROM SDADAM.M_DABRD_CMPG_CR_CLS  A
LEFT JOIN M_CMS_BZORG_CLS B
ON A.CLS_YYMM = B.CLS_YYMM AND A.CE_STFNO = B.STFNO
WHERE CMPG_MDCCD NOT IN ('B01','B02','B03','B04','B05','B06','B07',
						'D01','D02',
						'E01','E02','E04','E05')
GROUP BY A.CLS_YYMM, CASE WHEN TMNM LIKE '%POM%' THEN '[C02]POM'
		WHEN TMNM LIKE '%웹인%' THEN '[C01]웹인'
		WHEN TMNM LIKE '%미디어%' THEN '[C14]미디어'
		WHEN TMNM LIKE '%O/B%' THEN '[C07]O/B'
		WHEN TMNM LIKE '%혼합%' THEN '[C16]혼합'
		WHEN TMNM LIKE '%직인%' THEN '[C03]직인'
		WHEN TMNM LIKE '%호전환%' THEN '[C09]호전환'
		WHEN TMNM LIKE '%채팅%' THEN '[C08]채팅'
		WHEN TMNM LIKE '%인콜%' THEN '[C18]인콜'
		ELSE NULL END)
WHERE CMPG_MDCCD IS NOT NULL
ORDER BY CLS_YYMM , CMPG_MDCCD
;
