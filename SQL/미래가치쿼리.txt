DROP TABLE DATABIZ_FUTVAL_ALL_CR_TEMP_PRED PURGE;
CREATE TABLE DATABIZ_FUTVAL_ALL_CR_TEMP_PRED AS 
SELECT 
	A.*
	  ,CASE WHEN B.STFNO = '8901911'                       THEN '61'
	          WHEN B.STFNO = '6009607' THEN '11' /* 이관고객 전속으로 포함 */
            WHEN B.BZ_ATRCD IN ('30', '31')                THEN '62' /* LIFE WITH */
            WHEN B.BRNM LIKE '%LIFE%'                      THEN '62' /*LIFE WITH*/
            WHEN B.TMNM LIKE '%교차%'            THEN '12'  /* 생보형GA이외 교차는 교차로 배정 */
            WHEN B.BR_ORGCD IN ('1301406','1301908','1302142') THEN '12'
            WHEN B.TMNM LIKE '%한생GA팀%'                    THEN '20' /* 한생GA적용 */
            WHEN B.BZ_FML_QUFCD IN ('G01')                 THEN '20' /* 생보형GA(한생+미래에셋 GA) 적용 */
            WHEN B.BZ_ATRCD IN ('15','16','18','97','27')  THEN '12' /* ATCH_CRO_FLGCD 오류 보정 */
            WHEN B.BZ_FML_QUFCD IN ('39','47','56')        THEN '12' /* 교차사업소장, 교차매니저, 교차육성팀장  */
            WHEN B.BZ_PART_FLGCD = '06'
	            AND  COALESCE(B.BZ_ATRCD, '00') NOT IN ('15','16','18','97','27')
	            AND BRNM NOT LIKE '%TM%' THEN  '11'                  /*(24.11) TM이면 다이렉트로 배정 */
            WHEN B.BZ_PART_FLGCD = '06'
	            AND  B.BZ_FML_QUFCD IS NULL
	            AND BRNM NOT LIKE '%TM%'THEN  '11'                   /*(24.11) TM이면 다이렉트로 배정 */
            WHEN B.BRNM LIKE '%지역단%' THEN  '11'
            WHEN B.BRNM LIKE '%GA사업단%' THEN '20'
            WHEN B.BRNM LIKE '%HF%' THEN  '20'
            WHEN (B.HDQT_ORGCD IN ('1300011', '1302011','1303011'))               /* GA영업본부 */
            OR   (B.HDQT_ORGCD='1301318' AND B.BR_ORGCD='1301321')  /* 대구GA: 13.10~14.06 */
            OR   (B.HDQT_ORGCD='1300007' AND B.BR_ORGCD='1301040')  /* 중부GA: 13.10~14.06 */
            OR   (B.HDQT_ORGCD='1300008' AND B.BR_ORGCD='1301322')  /* 부산GA: 13.10~14.06 */
            OR   (B.BR_ORGCD = '1301409')                           /* 호남GA: 14.07~14.12 */
            OR   (B.BR_ORGCD = '1300163' AND B.BZP_ORGCD='1301383') /* 제주GA(가상): 14.02~ */
            OR   (B.BR_ORGCD = '1301040' AND B.BZP_ORGCD='1301396') /* 호남GA 영업 소파트 */
            OR   (B.BR_ORGCD = '1300090' AND B.BZP_ORGCD='1301384') /* 속초가상지점: 14.02 ~ 14.07 */  THEN '20'
            WHEN B.STFNO = '8066063'
            AND  B.BZP_ORGCD = '1301581'                                                          THEN '31' /* 20이후 사실상 폐쇄 */
            WHEN B.STFNO IN ('8066063', '8066059', '6010186', '3449868')                          THEN '32'
            WHEN B.HDQT_ORGCD IN ('1300013', '1300287','1300288')                                  THEN '40' /* 방카영업1-2부, 방카영업부: 18.12~ */
            WHEN B.HDQT_ORGCD IN ('1300012', '1301363','1303053') THEN '31'      /* 장기TM 영업부, 업무지원센터 */
            WHEN B.BR_ORGCD IN ('1300280', '1301363', '1303054', '1303055')    THEN '31'        /* 장기TM 영업부, 업무지원센터 */
            WHEN B.BR_ORGCD IN ('1300287', '1201452')                                             THEN '40' /* 방카영업부(김태현수정) */
            WHEN B.HDQT_ORGCD IN ('1301776', '1301362', '1301828') THEN '50'
            WHEN B.BZ_PART_FLGNM = '법인영업'
            AND  B.BZP_NM NOT LIKE '%신채널%'                                                       THEN '50'
            WHEN B.BZP_ORGCD IN ( '1301364','1301365','1301366','1301367','1301368'
                                , '1301369','1301370','1301371','1301372','1301373'
                                , '1301618','1301750','1301751','1301752','1301753'
                                , '1301754','1301755','1301756','1301757','1301758'
                                , '1301786','1301968','1302015','1302016')                        THEN '50'
            WHEN (
                 B.BZP_NM LIKE '%기업영업%'
            OR   B.BZP_NM LIKE '%법인영업%'
            OR   B.BRNM LIKE '%기업영업%'
            OR   B.BRNM LIKE '%법인영업%'
            OR   B.HDQNM LIKE '%기업영업%'
            OR   B.HDQNM LIKE '%법인영업%'
            OR   B.BRNM LIKE '%전략영업%'
            OR   B.BRNM LIKE '%GA영업%'
                 )                                                                                THEN '50'
            WHEN B.BZ_PART_FLGNM ='본사/기타'
            AND  B.BRNM NOT LIKE '%TM%'                                                           THEN '61'
            ELSE '61'
            END  AS CHNL
FROM DATABIZ_FUTVAL_YYMM_PRED T
LEFT JOIN M_CRM_MTHY_PS_CR A
ON 0=0
	AND A.CLS_YYMM = T.CLS_YYMM
LEFT JOIN M_ORG_MTHY_BZ_ORGN B
ON 0=0
	AND F_PREV_MONTH(T.CLS_YYMM,'-1') = B.CLS_YYMM
	AND A.DH_STFNO = B.STFNO;
	
	
DROP TABLE DATABIZ_FUTVAL_ALL_CUS_INFO_PRED PURGE;
CREATE TABLE DATABIZ_FUTVAL_ALL_CUS_INFO_PRED AS
SELECT 
	A.CLS_YYMM
	,C.CTMNO AS CTMNO  -- 현재 정상 계약의 계약자 OR 피보험자 정보
	, A.DH_STFNO
	,C.RELPC_TPCD AS RELPC_TPCD
	, CASE WHEN F_ISDATE(F_BIRTH_DAY_DEC(C.CTM_DSCNO)) = 'T' THEN TRUNC(MONTHS_BETWEEN(F_LAST_DAY(T.CLS_YYMM), TO_DATE(F_BIRTH_DAY_DEC(C.CTM_DSCNO),'YYYYMMDD')) / 12) 
				  ELSE NULL
				  END AS CTM_AGE
	,CASE WHEN F_SEX_DEC(C.CTM_DSCNO)='M' THEN '1.M'
			WHEN F_SEX_DEC(C.CTM_DSCNO)='F' THEN '2.F'
			ELSE NULL
			END AS CTM_SEX
	,MAX(CASE WHEN C.IKD_GRPCD = 'LA' THEN 1 ELSE 0 END) AS LA_YN
	,MAX(CASE WHEN C.IKD_GRPCD = 'CA' THEN 1 ELSE 0 END) AS CA_YN
	,MAX(CASE WHEN C.IKD_GRPCD IN ('FA','MA') THEN 1 ELSE 0 END) AS FAMA_YN
FROM DATABIZ_FUTVAL_ALL_CR_TEMP_PRED A
INNER JOIN DATABIZ_FUTVAL_YYMM_PRED T
ON 0=0
	AND A.CLS_YYMM = T.CLS_YYMM
INNER JOIN INS_CR_RELPC C
ON 0=0
	AND A.PLYNO = C.PLYNO
	AND C.NDS_AP_STR_DTHMS <= T.CLS_NDDT
	AND C.NDS_AP_ND_DTHMS > T.CLS_NDDT
	AND C.RELPC_TPCD IN ('01','02') /*관계자유형 = 계약자, 피보험자*/
	AND C.RELPC_STCD IN ('01','04') /*관계자상태코드 = 정상, 납입면제*/
GROUP BY 
	A.CLS_YYMM
	,C.CTMNO   -- 현재 정상 계약의 계약자 OR 피보험자 정보
	, A.DH_STFNO
	,C.RELPC_TPCD 
	, CASE WHEN F_ISDATE(F_BIRTH_DAY_DEC(C.CTM_DSCNO)) = 'T' THEN TRUNC(MONTHS_BETWEEN(F_LAST_DAY(T.CLS_YYMM), TO_DATE(F_BIRTH_DAY_DEC(C.CTM_DSCNO),'YYYYMMDD')) / 12) 
				  ELSE NULL
				  END 
	,CASE WHEN F_SEX_DEC(C.CTM_DSCNO)='M' THEN '1.M'
			WHEN F_SEX_DEC(C.CTM_DSCNO)='F' THEN '2.F'
			ELSE NULL
			END;

DROP TABLE DATABIZ_FUTVAL_CUS_ADD_PRED;
CREATE TABLE DATABIZ_FUTVAL_CUS_ADD_PRED AS
SELECT
  A1.CLS_YYMM
	, A1.CTMNO
	, CTM_SEX
	, CASE WHEN CTM_AGE BETWEEN 18 AND 29 THEN '20대 이하'
		   WHEN CTM_AGE BETWEEN 30 AND 39 THEN '30대'
		   WHEN CTM_AGE BETWEEN 40 AND 49 THEN '40대'
		   WHEN CTM_AGE BETWEEN 50 AND 59 THEN '50대'
		   WHEN CTM_AGE BETWEEN 60 AND 99 THEN '60대 이상'
		   ELSE NULL END AS AGE_CD
	, CTM_AGE
	, SD
	, SGNG
	, PSTNO
	, PSTNO_ID
	,E.JBCD
	,E.JB_CH_SEQNO
	,F.JBNM
FROM (
	SELECT DISTINCT 
		T.CLS_YYMM
		, T.CTMNO
		, T.CTM_SEX
		, T.CTM_AGE
		, A.LACO_DONG_FLGCD
		, A.BLDNM
		, A.SD
		, A.SGNG
		, A.PSTNO
		, A.REF_IT 
		, TO_CHAR(A.AP_STR_DTHMS,'YYYYMM')    AS AP_STR_YYMM
		, TO_CHAR(A.AP_ND_DTHMS,'YYYYMM')     AS AP_ND_YYMM
		, A.AP_STR_DTHMS
		, A.AP_ND_DTHMS
		, COALESCE(B.PSTNO_ID,C.PSTNO_ID, D.PSTNO_ID) AS PSTNO_ID
		, ROW_NUMBER() OVER(PARTITION BY T.CLS_YYMM, T.CTMNO ORDER BY A.AP_STR_DTHMS DESC, CASE WHEN B.PSTNO_ID IS NULL THEN 1 ELSE 0 END, CASE WHEN C.PSTNO_ID IS NULL THEN 1 ELSE 0 END) AS RK
	FROM DATABIZ_FUTVAL_ALL_CUS_INFO_PRED T
	INNER JOIN CUS_CTM_ADR_CRR A 
	ON 0=0
		AND T.CTMNO = A.CTMNO
		AND A.AP_ND_DTHMS > = F_LAST_DAY(T.CLS_YYMM) -- 시점
		AND A.AP_STR_DTHMS < F_LAST_DAY(T.CLS_YYMM) -- 시점
		AND A.ADR_FLGCD = '100' 
	
	LEFT OUTER JOIN CUS_PSTNO_RODNM_BA_ADR B 
	ON 1=1 
		AND A.BLD_ADMNO = B.BLD_ADMNO		
	
	LEFT OUTER JOIN CUS_PSTNO_RODNM_BA_ADR C
	ON 1=1 
		AND A.PSTNO = C.PSTNO
		AND A.SD=C.SD 
		AND A.SGNG = C.SGNG
		AND A.ROAD_NM = C.ROAD_NM
		AND A.MN_BLDNO = C.MN_BLDNO 
		AND COALESCE(A.SUB_BLDNO,'0') = COALESCE(C.SUB_BLDNO,'0') 
	
	LEFT OUTER JOIN CUS_PSTNO_RODNM_BA_ADR D
	ON 1=1 
		AND A.PSTNO = D.PSTNO
		AND A.SD=D.SD 
		AND A.SGNG = D.SGNG
		AND A.ROAD_NM = D.ROAD_NM
		AND A.MN_BLDNO = D.MN_BLDNO
	
	) A1
LEFT  JOIN CUS_CTM_JB_CRR E				
	ON 0=0			
		AND A1.CTMNO = E.CTMNO		
		AND E.AP_STR_DTHMS <= F_LAST_DAY(CLS_YYMM)
		AND E.AP_ND_DTHMS  >  F_LAST_DAY(CLS_YYMM)
		
	LEFT JOIN CUS_JB F		
	ON 0=0			
		AND E.JBCD = F.JBCD		
		AND E.JB_CH_SEQNO = F.JB_CH_SEQNO
WHERE 1=1
	AND RK = 1
;

	- 매핑 순서
	격자(성, 연령) -> 격자(ALL, 연령) -> 우편번호(성, 연령) -> 우편번호(ALL, 연령) -> 격자(성, ALL) -> 격자(ALL, ALL) -> 우편번호(성, ALL) -> 우편번호(ALL, ALL)
	*/
	SELECT
		B.*
		, CASE WHEN H.전문직 = 1 THEN COALESCE(D.SPJOB_AVG_ICAMT, D1.SPJOB_AVG_ICAMT, F.SPJOB_AVG_ICAMT, F1.SPJOB_AVG_ICAMT, D2.SPJOB_AVG_ICAMT, D3.SPJOB_AVG_ICAMT, F2.SPJOB_AVG_ICAMT, F3.SPJOB_AVG_ICAMT)
		       WHEN H.법인대표 = 1 THEN  COALESCE(D.CRP_PRS_AVG_ICAMT, D1.CRP_PRS_AVG_ICAMT, F.CRP_PRS_AVG_ICAMT, F1.CRP_PRS_AVG_ICAMT, D2.CRP_PRS_AVG_ICAMT, D3.CRP_PRS_AVG_ICAMT, F2.CRP_PRS_AVG_ICAMT, F3.CRP_PRS_AVG_ICAMT)
		       WHEN H.자영업 = 1 THEN  COALESCE(D.SFEMP_AVG_ICAMT, D1.SFEMP_AVG_ICAMT, F.SFEMP_AVG_ICAMT, F1.SFEMP_AVG_ICAMT, D2.SFEMP_AVG_ICAMT, D3.SFEMP_AVG_ICAMT, F2.SFEMP_AVG_ICAMT, F3.SFEMP_AVG_ICAMT)
		       WHEN H.주부 != 1 AND H.학생 != 1 AND H.무직 != 1 THEN  COALESCE(D.LBR_ICR_AVG_ICAMT, D1.LBR_ICR_AVG_ICAMT, F.LBR_ICR_AVG_ICAMT, F1.LBR_ICR_AVG_ICAMT, D2.LBR_ICR_AVG_ICAMT, D3.LBR_ICR_AVG_ICAMT, F2.LBR_ICR_AVG_ICAMT, F3.LBR_ICR_AVG_ICAMT)
		       ELSE NULL END AS INCOME_SEG /*평균소득 - 직업별*/
		, COALESCE(D.AVG_ICAMT, D1.AVG_ICAMT, F.AVG_ICAMT, F1.AVG_ICAMT, D2.AVG_ICAMT, D3.AVG_ICAMT, F2.AVG_ICAMT, F3.AVG_ICAMT) AS INCOME /*평균소득*/
		, CASE WHEN  H.주부 = 1 THEN COALESCE(D.HMKR_CRDVL_AVG_POCT, D1.HMKR_CRDVL_AVG_POCT, F.HMKR_CRDVL_AVG_POCT, F1.HMKR_CRDVL_AVG_POCT, D2.HMKR_CRDVL_AVG_POCT, D3.HMKR_CRDVL_AVG_POCT, F2.HMKR_CRDVL_AVG_POCT, F3.HMKR_CRDVL_AVG_POCT)
			   WHEN H.학생 = 1 THEN COALESCE(D.STT_CRDVL_AVG_POCT, D1.STT_CRDVL_AVG_POCT, F.STT_CRDVL_AVG_POCT, F1.STT_CRDVL_AVG_POCT, D2.STT_CRDVL_AVG_POCT, D3.STT_CRDVL_AVG_POCT, F2.STT_CRDVL_AVG_POCT, F3.STT_CRDVL_AVG_POCT)
			   WHEN H.무직 = 1 THEN COALESCE(D.OTJOB_CRDVL_AVG_POCT, D1.OTJOB_CRDVL_AVG_POCT, F.OTJOB_CRDVL_AVG_POCT, F1.OTJOB_CRDVL_AVG_POCT, D2.OTJOB_CRDVL_AVG_POCT, D3.OTJOB_CRDVL_AVG_POCT, F2.OTJOB_CRDVL_AVG_POCT, F3.OTJOB_CRDVL_AVG_POCT)
			   WHEN H.전문직 = 1 THEN COALESCE(D.SPJOB_CRDVL_AVG_POCT, D1.SPJOB_CRDVL_AVG_POCT, F.SPJOB_CRDVL_AVG_POCT, F1.SPJOB_CRDVL_AVG_POCT, D2.SPJOB_CRDVL_AVG_POCT, D3.SPJOB_CRDVL_AVG_POCT, F2.SPJOB_CRDVL_AVG_POCT, F3.SPJOB_CRDVL_AVG_POCT)
		       WHEN H.법인대표 = 1 THEN  COALESCE(D.CRP_PRS_CRDVL_AVG_POCT, D1.CRP_PRS_CRDVL_AVG_POCT, F.CRP_PRS_CRDVL_AVG_POCT, F1.CRP_PRS_CRDVL_AVG_POCT, D2.CRP_PRS_CRDVL_AVG_POCT, D3.CRP_PRS_CRDVL_AVG_POCT, F2.CRP_PRS_CRDVL_AVG_POCT, F3.CRP_PRS_CRDVL_AVG_POCT)
		       WHEN H.자영업 = 1 THEN  COALESCE(D.SFEMP_CRDVL_AVG_POCT, D1.SFEMP_CRDVL_AVG_POCT, F.SFEMP_CRDVL_AVG_POCT, F1.SFEMP_CRDVL_AVG_POCT, D2.SFEMP_CRDVL_AVG_POCT, D3.SFEMP_CRDVL_AVG_POCT, F2.SFEMP_CRDVL_AVG_POCT, F3.SFEMP_CRDVL_AVG_POCT)
		       ELSE NULL END AS CREDIT_SEG /*평균신용점수(NICE) - 직업별*/  	
		, COALESCE(D.AVG_CRD_VLPOC, D1.AVG_CRD_VLPOC, F.AVG_CRD_VLPOC, F1.AVG_CRD_VLPOC, D2.AVG_CRD_VLPOC, D3.AVG_CRD_VLPOC, F2.AVG_CRD_VLPOC, F3.AVG_CRD_VLPOC) AS CREDIT /*평균신용점수(NICE)*/
	FROM SDADAM.DATABIZ_FUTVAL_CUS_ADD_PRED B
    LEFT OUTER JOIN CUS_PSTNO_BAADR_SCT_ID C        /* 우편번호 ID별 250M격자 매칭값 */ 
    ON B.PSTNO_ID = C.PSTNO_ID
    LEFT OUTER JOIN M_CRM_IC_CRD_GRID_ST_STS D   /* 250M격자 기준 개인 단위 통계 */ 
    ON 0=0
        AND C.BAADR_SCT_ID = D.TOT_UNT_GRID_NM
        AND B.CTM_SEX = D.SEXCD
        AND B.AGE_CD = D.AGGB_NM 
        AND D.ST_YYMM = '202312'
    LEFT OUTER JOIN M_CRM_IC_CRD_GRID_ST_STS D1   /*  250M격자 기준 개인 단위 통계 - 성별ALL */
    ON 0=0
        AND C.BAADR_SCT_ID = D1.TOT_UNT_GRID_NM
        AND D1.SEXCD = 'ALL'
        AND B.AGE_CD = D1.AGGB_NM 
        AND D1.ST_YYMM = '202312'
    LEFT OUTER JOIN M_CRM_IC_CRD_GRID_ST_STS D2   /*  250M격자 기준 개인 단위 통계 - 연령ALL */ 
    ON 0=0
        AND C.BAADR_SCT_ID = D2.TOT_UNT_GRID_NM
        AND B.CTM_SEX = D2.SEXCD
        AND D2.AGGB_NM = 'ALL' 
        AND D2.ST_YYMM = '202312'
    LEFT OUTER JOIN M_CRM_IC_CRD_GRID_ST_STS D3   /*  250M격자 기준 개인 단위 통계 - 성별ALL, 연령ALL */ 
    ON 0=0
        AND C.BAADR_SCT_ID = D3.TOT_UNT_GRID_NM
        AND D3.SEXCD = 'ALL'
        AND D3.AGGB_NM = 'ALL'
        AND D3.ST_YYMM = '202312'
    ---
    LEFT OUTER JOIN M_CRM_IC_CRD_PSTNO_ST_STS F /* 우편번호 기준 개인 단위 통계 */ 
    ON 0=0
        AND B.PSTNO = F.TOT_UNT_PSTNO
        AND B.CTM_SEX = F.SEXCD
        AND B.AGE_CD = F.AGGB_NM
        AND F.ST_YYMM = '202312'
    LEFT OUTER JOIN M_CRM_IC_CRD_PSTNO_ST_STS F1 /*  우편번호 기준 개인 단위 통계 - 성별ALL */ 
    ON 0=0
        AND B.PSTNO = F1.TOT_UNT_PSTNO
        AND F1.SEXCD = 'ALL'
        AND B.AGE_CD = F1.AGGB_NM 
        AND F1.ST_YYMM = '202312'
    LEFT OUTER JOIN M_CRM_IC_CRD_PSTNO_ST_STS F2 /*  우편번호 기준 개인 단위 통계 - 연령ALL */ 
    ON 0=0
        AND B.PSTNO = F2.TOT_UNT_PSTNO
        AND B.CTM_SEX = F2.SEXCD
        AND F2.AGGB_NM = 'ALL'
        AND F2.ST_YYMM = '202312'
    LEFT OUTER JOIN M_CRM_IC_CRD_PSTNO_ST_STS F3 /*  우편번호 기준 개인 단위 통계 - 성별ALL, 연령ALL */ 
    ON 0=0
        AND B.PSTNO = F3.TOT_UNT_PSTNO
        AND F3.SEXCD = 'ALL'
        AND F3.AGGB_NM = 'ALL'
        AND F3.ST_YYMM = '202312' 
	LEFT OUTER JOIN (
		SELECT 
			CASE WHEN (
	            REGEXP_LIKE(A.JBNM, '판사|검사|변호사|법무사|변리사|노무사|회계사|세무사|감정평가사|관세사|의사|치과의사|수의사|한의사|약사|행정사|손해사정사|건축사|도선사|조종사|계리|교수|관제사')
	            OR
	            REGEXP_LIKE(A.JBNM, '전문 간호|항해사|선박 기관사|선장|고위|임원|경무관 이상|소방감 이상|금융 상품|자산|외환 딜러|소프트웨어|프로그래머|데이터|프로그래머|감독|장학관')
	            )
	        AND NOT REGEXP_LIKE(A.JBNM, '장의사|스포츠|건설|코치|검사원|검사공|검사기사|사육|판매|사무원|종사원|의료인|공무원|보조원|현장기술|감독자|제조|공급|사무및감독|경비원')	
			THEN 1 ELSE 0 END AS 전문직
			, CASE WHEN JBNM LIKE '%대표%'
				  AND JBCD LIKE '1%' THEN 1 /* 대표이사 */
				WHEN JBNM LIKE '%임원%' THEN 1
			    WHEN JBNM LIKE '%경영 및 진단 전문가%' THEN 1
			    WHEN JBNM LIKE '%경영인%' THEN 1
			    WHEN JBNM LIKE '%경영부서관리자%' THEN 1
				ELSE 0 END AS 법인대표 
			, CASE WHEN  JBNM LIKE '%자영업자%'      THEN 1
					 WHEN JBNM LIKE '%경영자%' 	      THEN 1

					 WHEN JBNM LIKE '%운영%'		  
					     AND JBNM NOT LIKE '%시스템%'
					     AND JBNM NOT LIKE '%전산%'
					     AND JBNM NOT LIKE '%정보%'
					     AND JBNM NOT LIKE '%기술%'
					     AND JBNM NOT LIKE '%연구%'
					     AND JBNM NOT LIKE '%사무%'
					 THEN 1
					 WHEN JBNM LIKE '%소매%'		  THEN 1
					 WHEN JBNM LIKE '%음식%'		  
					    AND JBNM LIKE '%서비스%' THEN 1
					 WHEN JBNM  LIKE '%조리%'		   
					  AND JBNM NOT LIKE '%산후%'  
					  AND JBNM NOT LIKE '%열차%'
					  AND JBNM NOT LIKE '%선박%'
					  AND JBNM NOT LIKE '%호텔%'
					  AND JBNM NOT LIKE '%병원%' 				THEN 1
					 WHEN JBNM  LIKE '%피부%'	    			THEN 1
					 WHEN JBNM  LIKE '%커피%'	    			THEN 1
					 WHEN JBNM LIKE '%식당 종사원%' 			THEN 1
					 WHEN JBNM  LIKE '%주방장%'	    			THEN 1
					 WHEN JBNM  LIKE '%미용사%'	    			THEN 1
					 WHEN JBNM  LIKE '%매장 계산원%'	        THEN 1
					 WHEN JBNM  LIKE '%음료서비스원%'	        THEN 1
					 WHEN JBNM  LIKE '%노래방%'	    			THEN 1
					 WHEN JBNM  LIKE '%자동차 경정비원%'	    THEN 1
					 WHEN JBNM  LIKE '%부동산 컨설턴트%'	    THEN 1
					 ELSE 0 END AS 자영업
			, CASE WHEN JBNM LIKE '%주부%' THEN 1 ELSE 0 END AS 주부
			, CASE WHEN JBNM LIKE '%학생%' THEN 1 ELSE 0 END AS 학생
			, CASE WHEN JBNM LIKE '%무직%' AND JBNM NOT LIKE '%사무직%' THEN 1 ELSE 0 END AS 무직
			, A.*
		FROM CUS_JB A
		WHERE 1=1
		AND LENGTH(JBCD) > 4
	) H
	ON 1=1 
	AND H.JBCD = B.JBCD
	AND H.JB_CH_SEQNO = B.JB_CH_SEQNO
);

DROP TABLE DATABIZ_FUTVAL_CUS_INFO_PRED PURGE;
CREATE TABLE DATABIZ_FUTVAL_CUS_INFO_PRED AS 
SELECT 
	A.*
	, MONTHS_BETWEEN(F_LAST_DAY(B.CLS_YYMM),HOME_ADR_CHDT)  HOME_ADR_CHDT
	, MONTHS_BETWEEN(F_LAST_DAY(B.CLS_YYMM),LATS_CR_CC_AGMDT)  LATS_CR_CC_AGMDT -- 계약체결동의일자
	, MONTHS_BETWEEN(F_LAST_DAY(B.CLS_YYMM),MKTG_COLUS_UTAGR_STRDT) MKTG_COLUS_UTAGR_STRDT -- 마동시작일자
	, MONTHS_BETWEEN(F_LAST_DAY(B.CLS_YYMM),MKTG_THRPS_UTAGR_STRDT)  MKTG_THRPS_UTAGR_STRDT-- 마케팅활용동의시작일자
	, MONTHS_BETWEEN(F_LAST_DAY(B.CLS_YYMM),LATS_CRNTC_MAIL_RCVDT) LATS_CRNTC_MAIL_RCVDT  -- 계약안내이메일수신일자
	
FROM DATABIZ_FUTVAL_ALL_CUS_INFO_PRED A
LEFT JOIN M_CRM_MTHY_PSN_CTM_INFO B
ON 0=0
	AND A.CTMNO = B.CTMNO
	AND A.CLS_YYMM = B.CLS_YYMM
;

	
DROP TABLE DATABIZ_FUTVAL_CR_CHNL_PRED;
CREATE TABLE DATABIZ_FUTVAL_CR_CHNL_PRED AS 
SELECT 
	*
FROM 
(SELECT 
	A.CLS_YYMM
	,C.CTMNO
	,A.CHNL
	,COUNT(DISTINCT A.PLYNO) PLYNO_CNT
FROM
(SELECT * FROM SDADAM.DATABIZ_FUTVAL_ALL_CR_TEMP_PRED
WHERE IKD_GRPCD = 'LA') A
INNER JOIN INS_CR_RELPC C
ON 0=0
  	AND A.PLYNO = C.PLYNO
  	AND C.NDS_AP_STR_DTHMS <= F_LAST_DAY(A.CLS_YYMM)
  	AND C.NDS_AP_ND_DTHMS > F_LAST_DAY(A.CLS_YYMM)
  	AND C.RELPC_TPCD IN ('02') /*관계자유형 = 계약자, 피보험자*/
  	AND C.RELPC_STCD IN ('01','04') /*관계자상태코드 = 정상, 납입면제*/
  	AND C.RELPC_DSCNO_FLGCD IN ('01', '05') -- 개인고객(법인고객 제외)
GROUP BY A.CLS_YYMM , C.CTMNO,A.CHNL) 
PIVOT
(
	MAX(PLYNO_CNT)
	FOR CHNL IN ('11' AS 전속,'12' AS 교차,'20' AS GA,'31' AS TM,'32' AS CM,'40' AS 방카,'50' AS 기업,'61' AS LIFEWITH,'62' AS 본사기타)
) P;
	
	
	
DROP TABLE DATABIZ_FUTVAL_CR_CRR_PRED;
CREATE TABLE DATABIZ_FUTVAL_CR_CRR_PRED AS
SELECT 
  CLS_YYMM
  , CTMNO
  , DH_STFNO
  , MAX(INS_ST) AS MX_INS
  , COUNT(DISTINCT CASE WHEN CR_STCD NOT IN ('00' , '12') THEN PLYNO ELSE NULL END) AS CCL_CR_CNT
FROM 
  (
  SELECT 
  	T.CLS_YYMM
  	, C.CTMNO
  	, A.PLYNO
  	, A.INS_ST
  	, D1.DH_STFNO 
  	, CR_STCD
  FROM DATABIZ_FUTVAL_YYMM_PRED T  
  INNER JOIN INS_INS_CR A
  ON 0=0
  	AND A.AP_STRDT <= T.CLS_NDDT_TS
  	AND A.AP_NDDT > T.CLS_NDDT_TS
  	AND A.INS_CLSTR > T.CLS_NDDT_TS
  	AND A.INS_ST <= T.CLS_NDDT_TS
  	AND IKD_GRPCD = 'LA'
  INNER JOIN INS_CR_ST_CRR B
  ON 0=0
  	AND A.PLYNO = B.PLYNO
  	AND B.NDS_AP_STR_DTHMS <= T.CLS_NDDT_TS
  	AND B.NDS_AP_ND_DTHMS > T.CLS_NDDT_TS
  INNER JOIN INS_CR_RELPC C
  ON 0=0
  	AND A.PLYNO = C.PLYNO
  	AND C.NDS_AP_STR_DTHMS <= T.CLS_NDDT_TS
  	AND C.NDS_AP_ND_DTHMS > T.CLS_NDDT_TS
  	AND C.RELPC_TPCD IN ('02') /*관계자유형 = 계약자, 피보험자*/
  	AND C.RELPC_STCD IN ('01','04') /*관계자상태코드 = 정상, 납입면제*/
  	AND C.RELPC_DSCNO_FLGCD IN ('01', '05') -- 개인고객(법인고객 제외)
  LEFT JOIN INS_CR_DH_STF_CRR D1
	ON 0=0
		AND A.PLYNO = D1.PLYNO
		AND D1.NDS_AP_STR_DTHMS <= T.CLS_NDDT_TS
		AND D1.NDS_AP_ND_DTHMS > T.CLS_NDDT_TS
		AND D1.DH_STF_TPCD = '01'
		AND D1.PRS_DH_STF_YN = '1'  
  WHERE EXISTS (SELECT 1 FROM DATABIZ_FUTVAL_ALL_CUS_INFO_PRED D WHERE D.CTMNO = C.CTMNO)
)
GROUP BY 
  CLS_YYMM
  , CTMNO
  , DH_STFNO
;

DROP TABLE DATABIZ_FUTVAL_PL_PRED PURGE;
CREATE TABLE DATABIZ_FUTVAL_PL_PRED  AS (	
SELECT *
FROM
(SELECT DISTINCT
	AA.CLS_YYMM --설계년월
	, AA.PLNO --설계번호
	, AA.PLDT --설계일자
	, AA.PL_STCD --설계상태코드
	, AA.PL_ST_CHDT -- 설계상태변경일자
	, AA.DH_STFNO AS CE_STFNO-- 취급직원번호
	, AA.USRNO AS CE_USRNO
	, BB.CTMNO -- 고개번호
	, BB.RELPC_TPCD -- 관계자유형코드
	, AA.PLNM
	, AA.GD_TYPE
	, AA.PL_FIN_CD
  , ROW_NUMBER() OVER (PARTITION BY AA.CLS_YYMM,AA.PLNO ORDER BY AA.PLDT DESC) AS RN
FROM (
	SELECT
		T.CLS_YYMM AS CLS_YYMM --설계년월
		, A.PLNO --설계번호
		, A.PLDT --설계일자
		, A.CGAF_CH_SEQNO -- 발행후변경순번
		, A.PL_STCD -- 설계상태코드
		, A.PL_ST_CHDT -- 설계상태변경일자
		, B.DH_STFNO -- 취급직원번호
		, B.USRNO
		, C.PLNM
		, CASE WHEN E.GDNM LIKE '%Young%' THEN 'L04' -- 자녀
           WHEN E.GDNM LIKE '%더건강 더실속%' OR E.MKTG_GD_CSFCD in ('L02','L01')					THEN 'L01' -- 종합
           WHEN E.GDNM LIKE '%시그니처 여성 건강%'	THEN 'L01'        -- 종합
      	   WHEN E.MKTG_GD_CSFCD = 'L03'							THEN 'L03' -- 운전자
      	   WHEN E.MKTG_GD_CSFCD = 'L04'							THEN 'L04' -- 자녀
      	   WHEN E.MKTG_GD_CSFCD = 'L06'							THEN 'L06' -- 암
      	   WHEN E.MKTG_GD_CSFCD = 'L11'							THEN 'L11' -- 재물
      	   WHEN E.MKTG_GD_CSFCD = 'L08'							THEN 'L08' -- 간병
      	   WHEN E.GDNM LIKE '%세이프투게더%'						THEN 'SAFE' -- 세이프투게더
           WHEN E.GDNM LIKE '%실손의료보험%'						THEN 'SS' -- 실손
      	   WHEN REGEXP_LIKE(E.GDNM, '참 편한|WELL100|간편|3N5|355')  THEN '유병자'
      	   WHEN E.MKTG_GD_CSFCD in ('L12','L13')				THEN 'L123' --연저축가입자수가 부족해서 기타로 분류
      	   ELSE 'ETC' END AS GD_TYPE
       , CASE WHEN A.PL_STCD IN ('03','04','05','06','07','08','09') THEN 1 ELSE 0 END AS PL_FIN_CD  -- 설계완료여부
	FROM DATABIZ_FUTVAL_YYMM_PRED T 
	INNER JOIN INS_INS_PL A
	  ON A.PLDT <= FIRST_DAY(T.CLS_YYMM)
	LEFT OUTER JOIN INS_PL_DH_STF  B    				/* 설계취급자 */
    ON   0=0
		    AND A.PLNO = B.PLNO
        AND A.CGAF_CH_SEQNO = B.CGAF_CH_SEQNO
        AND B.DH_STF_TPCD = '01'                 			/* 모집직원 */
        AND B.IKD_GRPCD = 'LA'
	LEFT JOIN INS_GD_CE_PL C
	ON 0=0
		AND A.PLCD = C.PLCD
		AND A.GDCD = C.GDCD
	LEFT JOIN M_CRM_GD_CSF_INFO E

SELECT 
  *
FROM
  (SELECT 
    CLS_YYMM
    ,CRT_CTMNO
    ,GD_TYPE
    ,COUNT(DISTINCT PLYNO) AS PLY_CNT
  FROM
    (
      SELECT 
        A.*
        , CASE WHEN GDNM LIKE '%Young%' THEN 'L04' -- 자녀
              WHEN GDNM LIKE '%더건강 더실속%' OR MKTG_GD_CSFCD in ('L02','L01')					THEN 'L01' -- 종합
              WHEN GDNM LIKE '%시그니처 여성 건강%'	THEN 'L01'        -- 종합
      	      WHEN MKTG_GD_CSFCD = 'L03'							THEN 'L03' -- 운전자
      	      WHEN MKTG_GD_CSFCD = 'L04'							THEN 'L04' -- 자녀
      	      WHEN MKTG_GD_CSFCD = 'L06'							THEN 'L06' -- 암
      	      WHEN MKTG_GD_CSFCD = 'L11'							THEN 'L11' -- 재물
      	      WHEN MKTG_GD_CSFCD = 'L08'							THEN 'L08' -- 간병
      	      WHEN GDNM LIKE '%세이프투게더%'						THEN 'SAFE' -- 세이프투게더
              WHEN GDNM LIKE '%실손의료보험%'						THEN 'SS' -- 실손
      	      WHEN REGEXP_LIKE(GDNM, '참 편한|WELL100|간편|3N5|355')  THEN '유병자'
      	      WHEN MKTG_GD_CSFCD in ('L12','L13')				THEN 'L123' --연저축가입자수가 부족해서 기타로 분류
      	   	  ELSE 'ETC' END AS GD_TYPE 
      FROM DATABIZ_FUTVAL_ALL_CR_TEMP_PRED A)
  GROUP BY 
    CLS_YYMM
    ,CRT_CTMNO
    ,GD_TYPE)
  PIVOT
  (
  	MAX(PLY_CNT)
  	FOR GD_TYPE IN ('L01' AS L01,'L03' AS L03,'L04' AS L04,'L06' AS L06,'L08' AS L08,'SAFE' AS SAFE , 'SS' AS SS, '유병자'  AS 유병자,'L11' AS L11,'L123' AS L123)
  ) P ;

        WHEN TRUNC(MONTHS_BETWEEN(F_LAST_DAY(A.CLS_YYMM),B9.MX_CLMDT),0) IS NULL THEN '이력없음'
        END AS MX_CLMDT	
  , CASE WHEN TRUNC(MONTHS_BETWEEN(F_LAST_DAY(A.CLS_YYMM),B10.MX_GNANL),0) >= 60 THEN '5년이상'
        WHEN TRUNC(MONTHS_BETWEEN(F_LAST_DAY(A.CLS_YYMM),B10.MX_GNANL),0) >= 48 THEN '4년이상'
        WHEN TRUNC(MONTHS_BETWEEN(F_LAST_DAY(A.CLS_YYMM),B10.MX_GNANL),0) >= 36 THEN '3년이상'
        WHEN TRUNC(MONTHS_BETWEEN(F_LAST_DAY(A.CLS_YYMM),B10.MX_GNANL),0) >= 24 THEN '2년이상'
        WHEN TRUNC(MONTHS_BETWEEN(F_LAST_DAY(A.CLS_YYMM),B10.MX_GNANL),0) >= 12 THEN '1년이상'
        WHEN TRUNC(MONTHS_BETWEEN(F_LAST_DAY(A.CLS_YYMM),B10.MX_GNANL),0) < 12 THEN '1년미만'
        WHEN TRUNC(MONTHS_BETWEEN(F_LAST_DAY(A.CLS_YYMM),B10.MX_GNANL),0) IS NULL THEN '이력없음'
        END AS MX_GNANL	
	, CASE WHEN B11.CTMNO IS NOT NULL THEN 1 ELSE 0 END AS SAME_ADD_YN
	, CASE WHEN B12.CTMNO IS NOT NULL THEN 1 ELSE 0 END AS FMY_CR_YN
	, CASE WHEN B13.CTMNO IS NOT NULL THEN 1 ELSE 0 END AS OTHER_CHNL_YN
	, CASE WHEN B15.PSS_CTM_CVDT IS NOT NULL THEN 1 ELSE 0 END AS PSS_CTM_YN
  , CASE WHEN TRUNC(MONTHS_BETWEEN(F_LAST_DAY(A.CLS_YYMM), CASE WHEN TO_CHAR(B15.INP_DTHMS,'YYYY') = '0001' THEN B15.MDF_DTHMS ELSE B15.INP_DTHMS END ),0) >= 60 THEN '5년이상'
        WHEN TRUNC(MONTHS_BETWEEN(F_LAST_DAY(A.CLS_YYMM), CASE WHEN TO_CHAR(B15.INP_DTHMS,'YYYY') = '0001' THEN B15.MDF_DTHMS ELSE B15.INP_DTHMS END ),0) >= 48 THEN '4년이상'
        WHEN TRUNC(MONTHS_BETWEEN(F_LAST_DAY(A.CLS_YYMM), CASE WHEN TO_CHAR(B15.INP_DTHMS,'YYYY') = '0001' THEN B15.MDF_DTHMS ELSE B15.INP_DTHMS END ),0) >= 36 THEN '3년이상'
        WHEN TRUNC(MONTHS_BETWEEN(F_LAST_DAY(A.CLS_YYMM), CASE WHEN TO_CHAR(B15.INP_DTHMS,'YYYY') = '0001' THEN B15.MDF_DTHMS ELSE B15.INP_DTHMS END ),0) >= 24 THEN '2년이상'
        WHEN TRUNC(MONTHS_BETWEEN(F_LAST_DAY(A.CLS_YYMM), CASE WHEN TO_CHAR(B15.INP_DTHMS,'YYYY') = '0001' THEN B15.MDF_DTHMS ELSE B15.INP_DTHMS END ),0) >= 12 THEN '1년이상'
        WHEN TRUNC(MONTHS_BETWEEN(F_LAST_DAY(A.CLS_YYMM), CASE WHEN TO_CHAR(B15.INP_DTHMS,'YYYY') = '0001' THEN B15.MDF_DTHMS ELSE B15.INP_DTHMS END ),0) < 12 THEN '1년미만'
        WHEN TRUNC(MONTHS_BETWEEN(F_LAST_DAY(A.CLS_YYMM), CASE WHEN TO_CHAR(B15.INP_DTHMS,'YYYY') = '0001' THEN B15.MDF_DTHMS ELSE B15.INP_DTHMS END ),0) IS NULL THEN '이력없음'
        END AS CUS_INP_DT		
  , CASE WHEN B16.ISPL_AGRE_YN = 1 THEN '동의'
        WHEN B16.ISPL_AGRE_YN = 0 AND TRUNC(MONTHS_BETWEEN(F_LAST_DAY(A.CLS_YYMM),B16.AGRE_NDDT),0) >= 60 THEN '5년이상'
        WHEN B16.ISPL_AGRE_YN = 0 AND TRUNC(MONTHS_BETWEEN(F_LAST_DAY(A.CLS_YYMM),B16.AGRE_NDDT),0) >= 48 THEN '4년이상'
        WHEN B16.ISPL_AGRE_YN = 0 AND TRUNC(MONTHS_BETWEEN(F_LAST_DAY(A.CLS_YYMM),B16.AGRE_NDDT),0) >= 36 THEN '3년이상'
        WHEN B16.ISPL_AGRE_YN = 0 AND TRUNC(MONTHS_BETWEEN(F_LAST_DAY(A.CLS_YYMM),B16.AGRE_NDDT),0) >= 24 THEN '2년이상'
        WHEN B16.ISPL_AGRE_YN = 0 AND TRUNC(MONTHS_BETWEEN(F_LAST_DAY(A.CLS_YYMM),B16.AGRE_NDDT),0) >= 12 THEN '1년이상'
        WHEN B16.ISPL_AGRE_YN = 0 AND TRUNC(MONTHS_BETWEEN(F_LAST_DAY(A.CLS_YYMM),B16.AGRE_NDDT),0) < 12 THEN '1년미만'
        ELSE '동의이력없음'
        END AS ISPL_AGRE	
	, NVL(PRM,0) PRM
  , NVL(B7.L01,0) L01
  , NVL(B7.L03,0) L03
  , NVL(B7.L04,0) L04
  , NVL(B7.L06,0)  L06
  , NVL(B7.L08,0) L08
  , NVL(B7.SAFE,0) SAFE
  , NVL(B7.SS,0) SS
  , NVL(B7.유병자,0) 유병자
  , NVL(B7.L11,0) L11
  , NVL(B7.L123,0) L123
	
FROM

DROP TABLE CRM_MART_ALL_CUS_INFO PURGE;
CREATE TABLE CRM_MART_ALL_CUS_INFO AS
SELECT 
	A.CLS_YYMM
	,C.CTMNO AS CTMNO  -- 현재 정상 계약의 계약자 OR 피보험자 정보
	,C.RELPC_TPCD AS RELPC_TPCD
	, CASE WHEN F_ISDATE(F_BIRTH_DAY_DEC(C.CTM_DSCNO)) = 'T' THEN TRUNC(MONTHS_BETWEEN(F_LAST_DAY(T.CLS_YYMM), TO_DATE(F_BIRTH_DAY_DEC(C.CTM_DSCNO),'YYYYMMDD')) / 12) 
				  ELSE NULL
				  END AS CTM_AGE
	,CASE WHEN F_SEX_DEC(C.CTM_DSCNO)='M' THEN '1.M'
			WHEN F_SEX_DEC(C.CTM_DSCNO)='F' THEN '2.F'
			ELSE NULL
			END AS CTM_SEX
	,MAX(CASE WHEN C.IKD_GRPCD = 'LA' THEN 1 ELSE 0 END) AS LA_YN
	,MAX(CASE WHEN C.IKD_GRPCD = 'CA' THEN 1 ELSE 0 END) AS CA_YN
	,MAX(CASE WHEN C.IKD_GRPCD IN ('FA','MA') THEN 1 ELSE 0 END) AS FAMA_YN
FROM CRM_MART_ALL_CR_TEMP A
INNER JOIN DATABIZ_FUTVAL_YYMM_PRED T
ON 0=0
	AND A.CLS_YYMM = T.CLS_YYMM
INNER JOIN INS_CR_RELPC C
ON 0=0
	AND A.PLYNO = C.PLYNO
	AND C.NDS_AP_STR_DTHMS <= T.CLS_NDDT
	AND C.NDS_AP_ND_DTHMS > T.CLS_NDDT
	AND C.RELPC_TPCD IN ('01','02') /*관계자유형 = 계약자, 피보험자*/
	AND C.RELPC_STCD IN ('01','04') /*관계자상태코드 = 정상, 납입면제*/
GROUP BY 
	A.CLS_YYMM
	,C.CTMNO   -- 현재 정상 계약의 계약자 OR 피보험자 정보
	,C.RELPC_TPCD 
	, CASE WHEN F_ISDATE(F_BIRTH_DAY_DEC(C.CTM_DSCNO)) = 'T' THEN TRUNC(MONTHS_BETWEEN(F_LAST_DAY(T.CLS_YYMM), TO_DATE(F_BIRTH_DAY_DEC(C.CTM_DSCNO),'YYYYMMDD')) / 12) 
				  ELSE NULL
				  END 
	,CASE WHEN F_SEX_DEC(C.CTM_DSCNO)='M' THEN '1.M'
			WHEN F_SEX_DEC(C.CTM_DSCNO)='F' THEN '2.F'
			ELSE NULL
			END;

	AND B.AP_NDDT > F_LAST_DAY(A.CLS_YYMM);



DROP TABLE SEG_MkTG_PRES_VAL PURGE;
CREATE TABLE SEG_MKTG_PRES_VAL AS 
SELECT
	DISTINCT
    A.CLS_YYMM
	, A.CTMNO
	, A.CTM_AGE
	, PLY_CNT
	, SM_LTRM_PRM
	, CASE WHEN A.CTM_AGE >= 70 THEN 70
		WHEN A.CTM_AGE >= 60 THEN 60
		WHEN A.CTM_AGE >= 50 THEN 50
		WHEN A.CTM_AGE >= 40 THEN 40
		WHEN A.CTM_AGE >= 30 THEN 30
		WHEN A.CTM_AGE >= 20 THEN 20
		WHEN A.CTM_AGE < 20 THEN 10
		ELSE NULL END AS CTM_AGE_SEG
FROM
(
    SELECT A.* 
    FROM DATABIZ_FUTVAL_YYMM_PRED T
    INNER JOIN SDADAM.CRM_MART_ALL_CUS_INFO A
    ON 0=0
        AND A.CLS_YYMM = T.CLS_YYMM
    WHERE 1=1
    	AND RELPC_TPCD = '02'
    	AND LA_YN = 1
	) A
LEFT JOIN 
	(
    SELECT 
	  A.CLS_YYMM
	  , C.CTMNO 
	  , COUNT(DISTINCT A.PLYNO) PLY_CNT  
	FROM DATABIZ_FUTVAL_YYMM_PRED T
    INNER JOIN SDADAM.CRM_MART_ALL_CR_TEMP A
    ON 0=0
        AND A.CLS_YYMM = T.CLS_YYMM
	INNER JOIN INS_CR_RELPC C
	ON 0=0
		AND A.PLYNO = C.PLYNO
		AND C.NDS_AP_STR_DTHMS <= F_LAST_DAY(A.CLS_YYMM)
		AND C.NDS_AP_ND_DTHMS > F_LAST_DAY(A.CLS_YYMM)
		AND C.RELPC_TPCD IN ('02') /*관계자유형 = 계약자, 피보험자*/
		AND C.RELPC_STCD IN ('01','04') /*관계자상태코드 = 정상, 납입면제*/
	WHERE A.IKD_GRPCD = 'LA'
		AND A.MKTG_GD_CSFCD NOT IN ('L12','L13') 
		AND A.GDNM NOT LIKE '%실손의료보험%'
	GROUP BY 
	  A.CLS_YYMM
	  , C.CTMNO
    ) B
ON 0=0
	AND A.CTMNO = B.CTMNO
    AND A.CLS_YYMM = B.CLS_YYMM
LEFT JOIN
	(
	SELECT 
        A.CLS_YYMM
		, C.CTMNO
		, SUM(A.LTRM_MPY_CV_PRM) AS SM_LTRM_PRM
	FROM DATABIZ_FUTVAL_YYMM_PRED T 
    INNER JOIN CRM_MART_LA_PRM A
    ON 0=0
        AND A.CLS_YYMM = T.CLS_YYMM
	INNER JOIN INS_CR_RELPC C
	ON 0=0
		AND A.PLYNO = C.PLYNO
		AND C.NDS_AP_STR_DTHMS <= F_LAST_DAY(A.CLS_YYMM)
		AND C.NDS_AP_ND_DTHMS > F_LAST_DAY(A.CLS_YYMM)
		AND C.RELPC_TPCD IN ('02') /*관계자유형 = 계약자, 피보험자*/
		AND C.RELPC_STCD IN ('01','04') /*관계자상태코드 = 정상, 납입면제*/
	WHERE 1=1
	  AND A.GD_TYPE NOT IN ('SS','L123')
	GROUP BY A.CLS_YYMM, C.CTMNO
	) D
ON 0=0
	AND A.CTMNO = D.CTMNO
    AND A.CLS_YYMM = D.CLS_YYMM

WHERE A.CTMNO NOT IN (SELECT
        	A.CTMNO
        FROM CUS_CTM A
        INNER JOIN M_ORG_MTHY_BZ_ORGN B
        ON 0=0
        	AND B.CLS_YYMM IN (SELECT CLS_YYMM FROM DATABIZ_FUTVAL_YYMM_PRED) /* 현재 */
        	AND B.RETDT >= SYSDATE 			/*퇴사자 제외 */
        	AND B.STF_BZ_STCD = '01' 									/* FP 상태 정상 */
        	AND A.CTM_DSCNO = B.RSNO)
	;
