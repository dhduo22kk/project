import pandas as pd
import re
import random
# --- 1) 상품명 클렌징 함수 ---------------------------------
def clean_product_name(name: str) -> str:
    if pd.isna(name):
        return ""
    name = re.sub(r'\([^)]*\)', '', name)
    keywords = ['무배당', '무배', '순수보장형',
                '2504', '2508', '2507', '2510', '2511', '2512','2509'
                ,'2601','2602','2603','2604','2605','2606','2607','2608','2609','2610','2611','2612']
    for kw in keywords:
        name = name.replace(kw, '')

    return name.strip()

# --- 2) 필터용 데이터 로딩 함수 ------------------------------
def load_filter_data(conn):
    # FILTER1 : 고지포기, 취소요청, 성별불명 고객
    sql1 = """
        SELECT * FROM GDREC_CTM_FILTER_1 WHERE CTMNO IS NOT NULL
    """
    df_filter1 = pd.read_sql(sql1, con=conn)

    # FILTER2 : 보장 충족 고객
    sql2 = """
    SELECT 
        A.*
        , CASE WHEN CV_CNT2 / CV_CNT >= 0.4 THEN 1 ELSE 0 END AS FILTER_YN
        , CV_CNT2 / CV_CNT RATIO 
    FROM GDREC_CTM_FILTER_2 A
    """
    df_filter2_raw = pd.read_sql(sql2, con=conn)

    # FILTER_YN = 1 : 아예 가입 불가 , FILTER_YN = 0 : 패널티만
    df_filter2 = df_filter2_raw.loc[df_filter2_raw['FILTER_YN'] == 1, ['CTMNO', 'GDCD']].reset_index(drop=True)
    df_penalty = df_filter2_raw.loc[df_filter2_raw['FILTER_YN'] == 0, ['CTMNO', 'GDCD', 'RATIO']].reset_index(drop=True)

    # 최종 “가입 불가” 필터 테이블 (FILTER1 + FILTER2=1)
    df_filter = pd.concat([df_filter1[['CTMNO', 'GDCD']], df_filter2], axis=0, ignore_index=True)
    # --- 상품 마스터 (판매 종료 상품 제외) --------------------
    sql3 = """
    SELECT * FROM GDREC_TMGD_LIST
    WHERE GDCD IN (SELECT GDCD FROM TB_LNG_BJONG_MASTER WHERE SALE_END_YN = 'Y')
    """
    df_GD = pd.read_sql(sql3, con=conn)

    # 상품명 클렌징
    df_GD['GDNM_CLEAN'] = df_GD['GDNM'].apply(clean_product_name)

    # 타입별 상품코드/상품명 매핑 사전
    type_to_gdcds = df_GD.groupby('GD_TYPE')['GDCD'].apply(list).to_dict()
    type_to_names = df_GD.groupby('GD_TYPE')['GDNM_CLEAN'].apply(lambda x: list(set(x))).to_dict()

    # ctno별 “가입불가 상품 리스트” 사전
    df_filter = df_filter.merge(df_GD[['GDCD','GD_TYPE','GDNM_CLEAN']] , on = 'GDCD' , how = 'left')
    df_filter = df_filter.rename(columns = {'CTMNO' : 'MN_NRDPS_CTMNO'})
    df_filter = df_filter.rename(columns = {'GD_TYPE' : 'product'})
    ctmno_block_dict = df_filter.groupby(['MN_NRDPS_CTMNO', 'product'])['GDNM_CLEAN'].apply(set).to_dict()
    
    return {
        "df_filter": df_filter,
        "df_penalty": df_penalty,
        "df_GD": df_GD,
        "TYPE_TO_GDCDS": type_to_gdcds,
        "TYPE_TO_NAMES": type_to_names,
        "CTMNO_BLOCK_DICT": ctmno_block_dict,
    }


def gd_filter_func(product,ctmage,ctmsex,customer):
    age = ctmage
    sex = ctmsex
    
    if product == 'L01':
        if age >= 60 and sex == '1.M':
            return '한화 굿밸런스종합보험'
        elif age >= 40 and sex == '1.M':
            return '한화 더건강한 한아름종합보험'
        elif age < 40 and sex == '1.M':
            return '한화 더건강한 1040 종합보험'
        elif age >= 20 and age >= 20 and sex == '2.F':
            return '한화 시그니처 여성 건강보험4.0'
        elif age < 20 and sex == '2.F':
            return '한화 더건강한 1040 종합보험'
    elif product == 'L04':
        if age >= 15:
            return '한화 더건강한 1040 종합보험'
        else: 
            return '한화 건강쑥쑥 어린이보험'
            
    elif product == 'L03':        
        return '한화 운전자상해보험'
            
    elif product == '유병자':
        si = customer[['입원청구', '입원고지','중대질환고지', '중대질환청구']]
        si = si.replace('X' , 99).astype(float)
        si_min = si.min()

        if int(si_min) == 99:
            return random.choice(['한화 3N5 더간편건강보험','한화 더 경증 간편건강보험Ⅱ'])
        elif int(si_min) == 0:
            return random.choice(['한화 3N5 더간편건강보험','한화 더 경증 간편건강보험Ⅱ'])
        elif int(si_min) > 0 and int(si_min) < 3:
            return '한화 311 간편건강보험'
        elif int(si_min) > 5:
            return '한화 더 경증 간편건강보험Ⅱ'
        elif int(si_min) >= 3 and sex == '1.M':
            return '한화 3N5 더간편건강보험'
        elif int(si_min) >= 3 and sex == '2.F':
            return random.choice(['한화 시그니처 여성 간편건강보험4.0'])
        
    elif product == 'SS':
        if age >= 50:
            return random.choice(['한화간편실손의료보험'])
        else: 
            return '한화실손의료보험'        
    else:
        if customer['gd_list']:
            return random.choice(customer['gd_list'])
        else: None

import pandas as pd
from datetime import datetime

def pivot_table(df):
    wide = df.pivot(
                index = 'CTMNO'
                , columns = 'RANK' 
                , values = ['상품코드','실제상품','확률','이유']
                )
    wide.columns = [f'추천{rank}_{col}' for col,rank in wide.columns]
    return wide.reset_index()

def custid_filter(conn , df):
    # FILTER : CUSTID가 존재하는 고객만
    sql = """
        SELECT 
            A.CUSTID
            ,B.CTMNO
        FROM TB_CUSTOMER A
        LEFT JOIN CUS_CTM B
        ON 0=0
            AND A.CUST_DSCNO = B.CTM_DSCNO
        WHERE B.CTMNO IN (SELECT DISTINCT CTMNO FROM GDREC_ORIGIN_INS_PRMCNT_PRED)
    """
    df_custid = pd.read_sql(sql,con=conn)
    df = df.merge(df_custid[['CTMNO','CUSTID']] , on = 'CTMNO' , how = 'inner')

    return df

def table_setting(df):
    df['REC_CD'] = '01'
    df['AP_SEQ'] = '1'
    df['AP_STRDT'] = datetime.now()
    df['AP_NDDT'] = datetime(9999,12,31)
    df['FIN_YN'] = '1'
    df1 = df[['CTMNO','REC_CD','AP_SEQ','AP_STRDT','AP_NDDT','추천1_상품코드','추천1_실제상품','추천1_확률','추천1_이유','CUSTID','FIN_YN']]
    df2 = df[['CTMNO','REC_CD','AP_SEQ','AP_STRDT','AP_NDDT','추천2_상품코드','추천2_실제상품','추천2_확률','추천2_이유','CUSTID','FIN_YN']]
    df3 = df[['CTMNO','REC_CD','AP_SEQ','AP_STRDT','AP_NDDT','추천3_상품코드','추천3_실제상품','추천3_확률','추천3_이유','CUSTID','FIN_YN']]
    df1.insert(loc=2, column = 'REC_RANK' , value = 1)
    df2.insert(loc=2, column = 'REC_RANK' , value = 2)
    df3.insert(loc=2, column = 'REC_RANK' , value = 3)
    lst = ['CTMNO','REC_CD','REC_RANK','AP_SEQ','AP_STRDT','AP_NDDT','REC_GDCD','REC_GDNM','REC_POCT','REC_RS1','CUSTID','FIN_YN']
    df1.columns , df2.columns , df3.columns = lst , lst , lst
    return pd.concat([df1,df2,df3],axis=0).reset_index(drop=True)

import os
from autogluon.tabular import TabularPredictor


BASE_DIR = '/notebooks/datajeonpart/상품추천시스템_V2/AutogluonModels'
PRODUCT_CODES = ['L01','L03','L04','L06','L08','SS','SI','L11']
PREDICTOR_PREFIXES =  ['predictor' , 'predictor2' , 'predictor3']



def load_predictor_paths(base_dir: str = BASE_DIR , prefixes = PREDICTOR_PREFIXES , codes = PRODUCT_CODES):
    paths  = {}
    for prefix in prefixes:
        for code in codes:
            filename = f'{prefix}_{code}.txt'
            full_path = os.path.join(base_dir , filename)

            if not os.path.exists(full_path):
                continue
            with open(full_path , 'r' , encoding = 'utf-8') as f:
                paths[(prefix , code)] = f.read().strip()
    return paths

def load_models():
    all_paths = load_predictor_paths()

    all_predictors = {}
    for (prefix , code) , path in all_paths.items():
        try:
            all_predictors[(prefix , code)] = TabularPredictor.load(path)
        except Exception as e:
            print(f'Error loading model for {prefix}_{code} at {path} : {e}')
    return all_predictors

def get_predictor(prefix : str , code : str , predictors):
    return predictors.get((prefix,code))
